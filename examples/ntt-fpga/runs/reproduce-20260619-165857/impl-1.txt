#!/usr/bin/env python3
"""
CFNTT Radix-2/4 NTT Multiplication Accelerator -- reverse-implemented from spec.

Functional model of the NTT datapath:
  * Barrett modular reduction (mu=21843, k=28) over q=12289
  * negacyclic forward NTT (CT/DIT) + inverse INTT (GS/DIF) with N^-1 scaling
  * radix-4 fused butterfly (== two radix-2 stages, j=psi^512 rotator)
  * conflict-free interleaved-BRAM bank mapping (XOR-fold)
  * minimal schedule FSM

Verification turns correctness into a handful of small finite obligations:
  - z3 proves Barrett reduction correct over the WHOLE input range [0, q^2)
  - z3 proves the bank map is conflict-free for every power-of-two stride and
    that (bank,offset) is a bijection on [0,N)
  - exhaustive check that the radix-4 butterfly == composed radix-2 over a field
  - O(N log N) round-trip + (small-N) negacyclic-convolution equivalence
  - structural FSM reachability
"""

import os
import sys
import random

try:
    import z3
except ImportError:
    print("FAIL: z3 is required (pip install z3-solver)")
    sys.exit(1)

# ---------------------------------------------------------------------------
# Authoritative parameters (from spec `metadata.spec.params` / widths)
# ---------------------------------------------------------------------------
Q           = 12289      # modulus
N           = 1024       # transform length
PSI         = 1945       # primitive 2048th root (negacyclic)
OMEGA       = 10302      # psi^2, N-th root
NINV        = 12277      # 1024^-1 mod q
BARRETT_MU  = 21843      # floor(2^28 / q)
BARRETT_K   = 28
G           = 11         # generator
COEFF_BITS  = 14
PROD_BITS   = 28
BANKS       = 8
PAR_BU      = 8

FAILURES = []
def check(cond, msg):
    if not cond:
        FAILURES.append(msg)
        print("FAIL: " + msg)

# ---------------------------------------------------------------------------
# Bit helpers
# ---------------------------------------------------------------------------
def bitrev(x, bits):
    r = 0
    for _ in range(bits):
        r = (r << 1) | (x & 1)
        x >>= 1
    return r

# ---------------------------------------------------------------------------
# Barrett reduction (hardware model: q-multiply then shift/subtract+correct)
# spec: t=(x*mu)>>k ; r=x-t*q ; <=2 conditional subtracts -> r in [0,q)
# ---------------------------------------------------------------------------
def barrett_reduce(x):
    assert 0 <= x < Q * Q, "Barrett input must be < q^2"
    t = (x * BARRETT_MU) >> BARRETT_K
    r = x - t * Q
    # at most two conditional subtractions land r in [0,q)
    if r >= Q:
        r -= Q
    if r >= Q:
        r -= Q
    return r

def mulmod(a, b):
    return barrett_reduce((a % Q) * (b % Q))

# ---------------------------------------------------------------------------
# Conflict-free bank mapping (committed XOR-fold map)
# ---------------------------------------------------------------------------
def bank(i):
    return (i & 7) ^ ((i >> 3) & 7) ^ ((i >> 6) & 7) ^ ((i >> 9) & 7)

def offset(i):
    return i >> 3

# ---------------------------------------------------------------------------
# Negacyclic NTT / INTT  (forward CT/DIT, inverse GS/DIF, merged psi-twist)
# ---------------------------------------------------------------------------
def make_tables(n, psi):
    bits = n.bit_length() - 1
    psi_pow = [pow(psi, i, Q) for i in range(n)]
    psi_inv = pow(psi, Q - 2, Q)
    psi_inv_pow = [pow(psi_inv, i, Q) for i in range(n)]
    fwd = [psi_pow[bitrev(i, bits)] for i in range(n)]
    inv = [psi_inv_pow[bitrev(i, bits)] for i in range(n)]
    return fwd, inv

def ntt(a, fwd):
    a = a[:]
    n = len(a)
    t = n
    m = 1
    while m < n:
        t //= 2
        for i in range(m):
            j1 = 2 * i * t
            S = fwd[m + i]
            for j in range(j1, j1 + t):
                U = a[j]
                V = mulmod(a[j + t], S)
                a[j] = (U + V) % Q
                a[j + t] = (U - V) % Q
        m *= 2
    return a

def intt(a, inv):
    a = a[:]
    n = len(a)
    t = 1
    m = n
    while m > 1:
        j1 = 0
        h = m // 2
        for i in range(h):
            S = inv[h + i]
            for j in range(j1, j1 + t):
                U = a[j]
                V = a[j + t]
                a[j] = (U + V) % Q
                a[j + t] = mulmod((U - V) % Q, S)
            j1 += 2 * t
        t *= 2
        m //= 2
    ninv = pow(n, Q - 2, Q)
    return [(x * ninv) % Q for x in a]

def negacyclic_conv(a, b):
    """Golden reference: product in Z_q[x]/(x^n + 1)."""
    n = len(a)
    r = [0] * n
    for i in range(n):
        ai = a[i]
        if ai == 0:
            continue
        for j in range(n):
            v = (ai * b[j]) % Q
            k = i + j
            if k < n:
                r[k] = (r[k] + v) % Q
            else:
                r[k - n] = (r[k - n] - v) % Q
    return [x % Q for x in r]

# ---------------------------------------------------------------------------
# Radix-4 fused butterfly (spec equations) vs composed radix-2 (golden)
# ---------------------------------------------------------------------------
def spec_radix4(x0, x1, x2, x3, w, j, p):
    w2 = (w * w) % p
    w3 = (w2 * w) % p
    b = (w * x1) % p
    c = (w2 * x2) % p
    d = (w3 * x3) % p
    y0 = (x0 + b + c + d) % p
    y1 = (x0 + j * b - c - j * d) % p
    y2 = (x0 - b + c - d) % p
    y3 = (x0 - j * b - c + j * d) % p
    return [y % p for y in (y0, y1, y2, y3)]

def golden_radix4(x0, x1, x2, x3, w, j, p):
    # two radix-2 stages: stage A twiddle w^2, stage B twiddles w / j*w
    w2 = (w * w) % p
    a0 = (x0 + w2 * x2) % p
    a1 = (x0 - w2 * x2) % p
    a2 = (x1 + w2 * x3) % p
    a3 = (x1 - w2 * x3) % p
    wa2 = (w * a2) % p
    wa3 = (w * a3) % p
    y0 = (a0 + wa2) % p
    y2 = (a0 - wa2) % p
    y1 = (a1 + j * wa3) % p
    y3 = (a1 - j * wa3) % p
    return [y % p for y in (y0, y1, y2, y3)]

def radix2_butterfly(x0, x1, w):
    u = x0 % Q
    v = mulmod(x1, w)
    return (u + v) % Q, (u - v) % Q

# ---------------------------------------------------------------------------
# Minimal schedule FSM
# ---------------------------------------------------------------------------
FSM_STATES = ["IDLE", "LOAD", "RUN_STAGES", "SCALE_INTT", "DRAIN", "DONE"]
FSM_EDGES = {
    "IDLE":       [("LOAD", "start")],
    "LOAD":       [("RUN_STAGES", "loaded")],
    "RUN_STAGES": [("RUN_STAGES", "stage<S"),
                   ("SCALE_INTT", "last & intt"),
                   ("DRAIN", "last & ntt")],
    "SCALE_INTT": [("DRAIN", "scaled")],
    "DRAIN":      [("DONE", "drained")],
    "DONE":       [("IDLE", "ack")],
}

def fsm_reachable(intt_mode):
    seen = {"IDLE"}
    stack = ["IDLE"]
    while stack:
        s = stack.pop()
        for (nxt, guard) in FSM_EDGES[s]:
            if not intt_mode and "intt" in guard:
                continue
            if intt_mode and guard == "last & ntt":
                continue
            if nxt not in seen:
                seen.add(nxt)
                stack.append(nxt)
    return "DONE" in seen

# ===========================================================================
# VERIFICATION
# ===========================================================================
def verify_constants():
    check(pow(PSI, 2048, Q) == 1, "psi^2048 != 1")
    check(pow(PSI, 1024, Q) == Q - 1, "psi^1024 != -1 (not negacyclic)")
    check(pow(PSI, 512, Q) * pow(PSI, 512, Q) % Q == Q - 1, "j=psi^512: j^2 != -1")
    check((PSI * PSI) % Q == OMEGA, "omega != psi^2")
    check(pow(N, Q - 2, Q) == NINV, "Ninv != 1024^-1 mod q")
    check(BARRETT_MU == (1 << BARRETT_K) // Q, "barrett mu mismatch")
    check(Q * Q < (1 << BARRETT_K), "q^2 must be < 2^k for the Barrett bound")
    check(pow(G, 6, Q) == PSI, "g^6 != psi")
    check((Q - 1) % (2 * N) == 0, "2N does not divide q-1")
    # op-count claims (within-radix-4): 33% mult, 20% add/sub
    check(round((1 - 2 / 3) * 100) == 33, "radix-4 mult saving != 33%")
    check(round((1 - 8 / 10) * 100) == 20, "radix-4 add/sub saving != 20%")

def verify_barrett_z3():
    """Prove barrett_reduce(x) lands in [0,q) for every x in [0, q^2)."""
    x = z3.Int('x')
    t = (x * BARRETT_MU) / (1 << BARRETT_K)        # z3 Int division = floor for x>=0
    r0 = x - t * Q
    r1 = z3.If(r0 >= Q, r0 - Q, r0)
    r2 = z3.If(r1 >= Q, r1 - Q, r1)
    s = z3.Solver()
    s.add(x >= 0, x < Q * Q)
    s.add(z3.Or(r2 < 0, r2 >= Q))                  # seek a violation of the range
    res = s.check()
    if res == z3.sat:
        m = s.model()
        check(False, "Barrett out of [0,q): x=%s" % m[x])
    else:
        check(res == z3.unsat, "Barrett z3 inconclusive: %s" % res)
    # congruence is automatic (only multiples of q subtracted); spot-check it
    for xv in [0, 1, Q - 1, Q, Q * Q - 1, 151019520, 268428627 % (Q * Q),
               12288 * 12288, 70000000, 12289 * 12289 - 1]:
        check(barrett_reduce(xv) == xv % Q, "Barrett value wrong at x=%d" % xv)

def _z3_bank(i):
    return (i & 7) ^ (z3.LShR(i, 3) & 7) ^ (z3.LShR(i, 6) & 7) ^ (z3.LShR(i, 9) & 7)

def verify_bank_conflict_free():
    """Radix-2: operands (i, i^2^s) always in distinct banks, every stride s."""
    for s in range(10):
        i = z3.BitVec('i', 10)
        partner = i ^ z3.BitVecVal(1 << s, 10)
        sol = z3.Solver()
        sol.add(_z3_bank(i) == _z3_bank(partner))
        r = sol.check()
        if r == z3.sat:
            m = sol.model()
            check(False, "radix-2 bank collision at stride 2^%d, i=%s" % (s, m[i]))
        else:
            check(r == z3.unsat, "bank conflict z3 inconclusive (stride %d)" % s)

    # Radix-4: the four operands (i, i^2^s, i^2^(s+1), i^2^s^2^(s+1)) distinct banks
    for s in range(9):
        i = z3.BitVec('i', 10)
        a = i
        b = i ^ z3.BitVecVal(1 << s, 10)
        c = i ^ z3.BitVecVal(1 << (s + 1), 10)
        d = i ^ z3.BitVecVal((1 << s) | (1 << (s + 1)), 10)
        banks = [_z3_bank(x) for x in (a, b, c, d)]
        pair_eq = z3.Or([banks[p] == banks[q]
                         for p in range(4) for q in range(p + 1, 4)])
        sol = z3.Solver()
        sol.add(pair_eq)
        r = sol.check()
        if r == z3.sat:
            m = sol.model()
            check(False, "radix-4 bank collision at stride 2^%d, i=%s" % (s, m[i]))
        else:
            check(r == z3.unsat, "radix-4 bank z3 inconclusive (stride %d)" % s)

def verify_bank_bijection():
    """(bank,offset) is injective on [0,N): no two indices share both."""
    i = z3.BitVec('i', 10)
    j = z3.BitVec('j', 10)
    sol = z3.Solver()
    sol.add(i != j)
    sol.add(_z3_bank(i) == _z3_bank(j))
    sol.add(z3.LShR(i, 3) == z3.LShR(j, 3))   # equal offsets
    r = sol.check()
    if r == z3.sat:
        m = sol.model()
        check(False, "bank/offset collision: i=%s j=%s" % (m[i], m[j]))
    else:
        check(r == z3.unsat, "bijection z3 inconclusive: %s" % r)

def verify_radix4_butterfly():
    """Exhaustively: spec radix-4 == composed radix-2 over a small field."""
    p = 5
    j = 2                                     # 2^2 = 4 = -1 mod 5
    rng = range(p)
    for x0 in rng:
        for x1 in rng:
            for x2 in rng:
                for x3 in rng:
                    for w in rng:
                        sp = spec_radix4(x0, x1, x2, x3, w, j, p)
                        go = golden_radix4(x0, x1, x2, x3, w, j, p)
                        if sp != go:
                            check(False, "radix4 mismatch x=%s w=%d: %s vs %s"
                                  % ((x0, x1, x2, x3), w, sp, go))
                            return
    # also confirm with the real field/rotator on random samples
    jq = pow(PSI, 512, Q)
    for _ in range(2000):
        xs = [random.randrange(Q) for _ in range(4)]
        w = random.randrange(Q)
        if spec_radix4(*xs, w, jq, Q) != golden_radix4(*xs, w, jq, Q):
            check(False, "radix4 mismatch (real field) xs=%s w=%d" % (xs, w))
            return

def verify_radix2_butterfly_z3():
    """One radix-2 butterfly: outputs equal (u+v) mod q and (u-v) mod q."""
    u = z3.Int('u'); v = z3.Int('v')
    y0 = z3.Int('y0'); y1 = z3.Int('y1')
    s = z3.Solver()
    s.add(u >= 0, u < Q, v >= 0, v < Q)
    s.add(y0 == z3.If(u + v >= Q, u + v - Q, u + v))
    s.add(y1 == z3.If(u - v < 0, u - v + Q, u - v))
    s.add(z3.Or(z3.Or(y0 < 0, y0 >= Q),
                z3.Or(y1 < 0, y1 >= Q),
                (y0 - (u + v)) % Q != 0,
                (y1 - (u - v)) % Q != 0))
    r = s.check()
    check(r == z3.unsat, "radix-2 butterfly z3 found a violation: %s" % r)

def verify_transform_roundtrip():
    """Full N=1024 negacyclic NTT/INTT round-trip identity (O(N log N))."""
    fwd, inv = make_tables(N, PSI)
    # the table-derived Ninv must equal the spec value
    check(pow(N, Q - 2, Q) == NINV, "round-trip Ninv mismatch")
    for _ in range(5):
        a = [random.randrange(Q) for _ in range(N)]
        out = intt(ntt(a, fwd), inv)
        if out != a:
            bad = next(k for k in range(N) if out[k] != a[k])
            check(False, "INTT(NTT(a)) != a at index %d (%d vs %d)"
                  % (bad, out[bad], a[bad]))
            return

def verify_negacyclic_convolution(n):
    """NTT-domain pointwise product == polynomial mult mod (x^n+1)."""
    psi_n = pow(G, 12288 // (2 * n), Q)
    check(pow(psi_n, n, Q) == Q - 1, "psi_%d not negacyclic" % n)
    fwd, inv = make_tables(n, psi_n)
    for _ in range(3):
        a = [random.randrange(Q) for _ in range(n)]
        b = [random.randrange(Q) for _ in range(n)]
        fa = ntt(a, fwd)
        fb = ntt(b, fwd)
        prod = [mulmod(fa[i], fb[i]) for i in range(n)]
        got = intt(prod, inv)
        want = negacyclic_conv(a, b)
        if got != want:
            bad = next(k for k in range(n) if got[k] != want[k])
            check(False, "negacyclic conv mismatch at %d (%d vs %d)"
                  % (bad, got[bad], want[bad]))
            return

def verify_intt_scale():
    """The INTT scaling unit: per-coeff multiply by Ninv then Barrett-reduce."""
    for x in [0, 1, Q - 1, 12288, 7777, 4242]:
        check(mulmod(x, NINV) == (x * NINV) % Q, "intt_scale wrong at %d" % x)

def verify_fsm():
    check(fsm_reachable(intt_mode=False), "FSM: DONE unreachable in NTT mode")
    check(fsm_reachable(intt_mode=True),  "FSM: DONE unreachable in INTT mode")
    check(set(FSM_EDGES.keys()) == set(FSM_STATES), "FSM state set mismatch")

def verify_widths():
    check(COEFF_BITS == 14, "coefficient width != 14")
    check(PROD_BITS == 28 and PROD_BITS == 2 * COEFF_BITS, "product width != 28")
    check((Q - 1).bit_length() == COEFF_BITS, "q does not fit in 14 bits")
    check(BANKS == 8 and PAR_BU == 8, "banks / parallel BU != 8")
    check(N % BANKS == 0 and N // BANKS == 128, "depth per bank != N/banks=128")

# ===========================================================================
def main():
    random.seed(0xCF77)
    verify_constants()
    verify_widths()
    verify_barrett_z3()
    verify_radix2_butterfly_z3()
    verify_bank_conflict_free()
    verify_bank_bijection()
    verify_radix4_butterfly()
    verify_intt_scale()
    verify_fsm()
    verify_transform_roundtrip()
    verify_negacyclic_convolution(256)

    # Optional production-size convolution (O(N^2) golden): off by default.
    if os.environ.get("DEEP_VERIFY") == "1":
        verify_negacyclic_convolution(1024)

    if FAILURES:
        sys.exit(1)
    print("VERIFIED")
    sys.exit(0)

if __name__ == "__main__":
    main()
