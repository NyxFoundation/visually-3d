#!/usr/bin/env python3
"""
Self-contained reproduction + self-check of the CFNTT radix-2/4 NTT accelerator.

Implemented from the scene spec ALONE (functional fields only):
  - Barrett modular reduction (mod_red_mult + mod_red_subshift)
  - conflict-free interleaved-bank memory map (crossbar_network / addr_gen_unit)
  - merged-negacyclic radix-2 NTT/INTT datapath (butterfly array + intt_scale)
  - radix-4 fused butterfly (the four fusion yokes / exploded BU)

Verification turns correctness into small finite obligations and discharges each
with the cheapest sound check (z3 over bounded domains, exhaustive small spaces,
O(N log N) round-trips). Prints VERIFIED / FAIL.
"""
import os
import sys
import random
import z3

# ---- committed parameters (spec: metadata.spec.params / widths) ----
Q          = 12289        # 14-bit NTT-friendly prime, q = 3*2^12 + 1
N          = 1024         # transform length
PSI        = 1945         # primitive 2N-th root (negacyclic), order 2048
OMEGA      = 10302        # psi^2, N-th root
NINV       = 12277        # 1024^-1 mod q
GEN        = 11           # generator g, psi = g^6
MU         = 21843        # Barrett mu = floor(2^28 / q)
KSH        = 28           # Barrett shift k
COEFF_BITS = 14
PROD_BITS  = 28
PAR_BU     = 8
J          = pow(PSI, 512, Q)   # constant radix-4 rotator, J^2 == -1


def fail(msg):
    print("FAIL: " + msg)
    sys.exit(1)


# ======================================================================
# Implementation (plain Python functions)
# ======================================================================

# ---- Barrett reduction: estimate then <=2 conditional subtractions ----
def barrett(x):                      # x in [0, q^2)
    qh = (x * MU) >> KSH
    r = x - qh * Q
    if r >= Q:
        r -= Q
    if r >= Q:
        r -= Q
    return r


def mulmod(a, b):                    # a,b in [0,Q)
    return barrett(a * b)


def addmod(a, b):
    return (a + b) % Q


def submod(a, b):
    return (a - b) % Q


# ---- conflict-free memory map (committed XOR-fold) ----
def bank(i):
    return (i & 7) ^ ((i >> 3) & 7) ^ ((i >> 6) & 7) ^ ((i >> 9) & 7)


def offset(i):
    return i >> 3


# ---- bit reversal ----
def brv(x, bits):
    r = 0
    for _ in range(bits):
        r = (r << 1) | (x & 1)
        x >>= 1
    return r


# ---- twiddle tables for a given size ----
def root_for(n):
    # PSI has order 2048; PSI^(1024/n) is a primitive 2n-th root mod Q
    return pow(PSI, N // n, Q)


def build_tables(n):
    psi_n = root_for(n)
    logn = n.bit_length() - 1
    psi_inv = pow(psi_n, Q - 2, Q)
    zetas = [pow(psi_n, brv(i, logn), Q) for i in range(n)]
    zetas_inv = [pow(psi_inv, brv(i, logn), Q) for i in range(n)]
    return psi_n, zetas, zetas_inv


# ---- radix-2 forward NTT (CT/DIT, merged negacyclic; bit-reversed output) ----
def ntt(a, zetas):
    a = a[:]
    n = len(a)
    t = n
    m = 1
    while m < n:
        t //= 2
        for i in range(m):
            S = zetas[m + i]
            base = 2 * i * t
            for j in range(base, base + t):
                U = a[j]
                V = mulmod(a[j + t], S)
                a[j] = addmod(U, V)
                a[j + t] = submod(U, V)
        m *= 2
    return a


# ---- radix-2 inverse NTT (GS/DIF; bit-reversed input, natural output) ----
def intt(a, zetas_inv):
    a = a[:]
    n = len(a)
    ninv = pow(n, Q - 2, Q)
    t = 1
    m = n
    while m > 1:
        h = m // 2
        j1 = 0
        for i in range(h):
            S = zetas_inv[h + i]
            for j in range(j1, j1 + t):
                U = a[j]
                V = a[j + t]
                a[j] = addmod(U, V)
                a[j + t] = mulmod(submod(U, V), S)
            j1 += 2 * t
        t *= 2
        m //= 2
    return [mulmod(x, ninv) for x in a]


# ---- radix-4 fused butterfly (DFT4 of twiddled inputs) ----
def radix4_bf(x0, x1, x2, x3, w):
    w2 = mulmod(w, w)
    w3 = mulmod(w2, w)
    b = mulmod(w, x1)
    c = mulmod(w2, x2)
    d = mulmod(w3, x3)
    jb = mulmod(J, b)
    jd = mulmod(J, d)
    y0 = (x0 + b + c + d) % Q
    y1 = (x0 + jb - c - jd) % Q
    y2 = (x0 - b + c - d) % Q
    y3 = (x0 - jb - c + jd) % Q
    return y0, y1, y2, y3


def twostage_bf(x0, x1, x2, x3, w):
    # the architectural claim: one radix-4 BU == two radix-2 stages on 4 points
    w2 = mulmod(w, w)
    w3 = mulmod(w2, w)
    b = mulmod(w, x1)
    c = mulmod(w2, x2)
    d = mulmod(w3, x3)
    a0 = (x0 + c) % Q
    a1 = (x0 - c) % Q
    a2 = (b + d) % Q
    a3 = (b - d) % Q
    ja3 = mulmod(J, a3)
    Y0 = (a0 + a2) % Q
    Y2 = (a0 - a2) % Q
    Y1 = (a1 + ja3) % Q
    Y3 = (a1 - ja3) % Q
    return Y0, Y1, Y2, Y3


# ======================================================================
# Independent golden / reference models (derived from the spec definition)
# ======================================================================

def golden_ntt(a, psi_n):
    # negacyclic transform, natural order: X[k] = a(psi_n^(2k+1)) = sum a_j psi^j omega^(jk)
    n = len(a)
    om = mulmod(psi_n, psi_n)
    out = []
    for k in range(n):
        s = 0
        for j in range(n):
            term = a[j] * pow(psi_n, j, Q) % Q
            term = term * pow(om, (j * k) % n, Q) % Q
            s = (s + term) % Q
        out.append(s)
    return out


def negmul(a, b):
    # schoolbook negacyclic convolution mod (x^n + 1) over GF(Q)
    n = len(a)
    c = [0] * n
    for i in range(n):
        ai = a[i]
        if ai == 0:
            continue
        for j in range(n):
            v = mulmod(ai, b[j])
            k = i + j
            if k < n:
                c[k] = (c[k] + v) % Q
            else:
                c[k - n] = (c[k - n] - v) % Q
    return c

# Verification obligations (fast, finite, sound)
# ======================================================================

def check_constants():
    bad = []

    def req(cond, why):
        if not cond:
            bad.append(why)

    req(pow(GEN, 6, Q) == PSI, "psi != g^6")
    req(pow(PSI, 2, Q) == OMEGA, "omega != psi^2")
    req(pow(PSI, 2048, Q) == 1, "psi^2048 != 1")
    req(pow(PSI, 1024, Q) == Q - 1, "psi^1024 != -1 (not negacyclic)")
    req(pow(OMEGA, 1024, Q) == 1, "omega^1024 != 1")
    req(pow(OMEGA, 512, Q) == Q - 1, "omega^512 != -1")
    req(1024 * NINV % Q == 1, "Ninv != 1024^-1 mod q")
    req(pow(J, 2, Q) == Q - 1, "J^2 != -1 (bad radix-4 rotator)")
    req(MU == (1 << KSH) // Q, "mu != floor(2^28/q)")
    req(Q * Q < (1 << KSH), "q^2 not < 2^28 (Barrett input bound broken)")
    req(Q < (1 << COEFF_BITS), "q does not fit 14-bit coefficient width")
    req(Q * Q < (1 << PROD_BITS), "q^2 does not fit 28-bit product width")
    req(3 + 1 + 4 + 1 == 9, "pipeline latency mult3+add1+barrett4+reg1 != 9")
    req(N // 2 // PAR_BU == 64, "radix-2 bf cycles/stage != 64")
    req(10 * 64 == 640, "radix-2 total bf cycles != 640")
    req(N // 4 // PAR_BU == 32, "radix-4 bf cycles/stage != 32")
    req(5 * 32 == 160, "radix-4 total bf cycles != 160")
    if bad:
        fail("constants: " + "; ".join(bad))


def prove_barrett():
    # for all x in [0, q^2): the 2-subtraction Barrett path == x mod q, in [0,q)
    x = z3.Int('x')
    qh = (x * MU) / (1 << KSH)        # z3 Int division == floor for x>=0
    r0 = x - qh * Q
    r1 = z3.If(r0 >= Q, r0 - Q, r0)
    r2 = z3.If(r1 >= Q, r1 - Q, r1)
    s = z3.Solver()
    s.add(x >= 0, x < Q * Q)
    s.add(z3.Or(r2 != x % Q, r2 < 0, r2 >= Q))
    if s.check() != z3.unsat:
        fail("Barrett reduction wrong, counterexample x=" + str(s.model()))


def _bank_z3(i):
    return (i & 7) ^ (z3.LShR(i, 3) & 7) ^ (z3.LShR(i, 6) & 7) ^ (z3.LShR(i, 9) & 7)


def prove_conflict_free():
    W = 16
    # radix-2: operands (i, i+2^s) with bit s clear land in distinct banks
    for s in range(10):
        i = z3.BitVec('i', W)
        bit = z3.BitVecVal(1 << s, W)
        sol = z3.Solver()
        sol.add(z3.ULT(i, N))
        sol.add((z3.LShR(i, s) & 1) == 0)
        sol.add(_bank_z3(i) == _bank_z3(i | bit))
        if sol.check() != z3.unsat:
            fail("radix-2 bank conflict at stride 2^%d" % s)
    # radix-4: operands (i, i+2^s, i+2^{s+1}, i+3*2^s) all in distinct banks
    for s in range(9):
        i = z3.BitVec('i', W)
        b0 = z3.BitVecVal(1 << s, W)
        b1 = z3.BitVecVal(1 << (s + 1), W)
        idx = [i, i | b0, i | b1, i | b0 | b1]
        sol = z3.Solver()
        sol.add(z3.ULT(i, N))
        sol.add((z3.LShR(i, s) & 1) == 0)
        sol.add((z3.LShR(i, s + 1) & 1) == 0)
        collide = []
        for a in range(4):
            for b in range(a + 1, 4):
                collide.append(_bank_z3(idx[a]) == _bank_z3(idx[b]))
        sol.add(z3.Or(*collide))
        if sol.check() != z3.unsat:
            fail("radix-4 bank conflict at stride 2^%d" % s)


def check_bijection():
    # (bank, offset) must be a bijection [0,N) -> 8 banks x 128 rows
    seen = set()
    for i in range(N):
        key = (bank(i), offset(i))
        if key in seen:
            fail("(bank,offset) collision at i=%d" % i)
        seen.add(key)
    if len(seen) != N:
        fail("(bank,offset) is not a bijection over [0,N)")
    if max(offset(i) for i in range(N)) != 127:
        fail("offset range not [0,127] (128 words/bank)")


def prove_radix4_fusion():
    # symbolic: radix-4 DFT4(x0,b,c,d) == two radix-2 stages, identically (as ints)
    x0, b, c, d, j = z3.Ints('x0 b c d j')
    y0 = x0 + b + c + d
    y1 = x0 + j * b - c - j * d
    y2 = x0 - b + c - d
    y3 = x0 - j * b - c + j * d
    a0, a1, a2, a3 = x0 + c, x0 - c, b + d, b - d
    Y0, Y2 = a0 + a2, a0 - a2
    Y1, Y3 = a1 + j * a3, a1 - j * a3
    sol = z3.Solver()
    sol.add(z3.Or(y0 != Y0, y1 != Y1, y2 != Y2, y3 != Y3))
    if sol.check() != z3.unsat:
        fail("radix-4 butterfly != two radix-2 stages")


def check_radix4_numeric():
    # concrete mod-q confirmation of the fusion claim
    rnd = random.Random(7)
    for _ in range(3000):
        x0, x1, x2, x3 = (rnd.randrange(Q) for _ in range(4))
        w = rnd.randrange(Q)
        if radix4_bf(x0, x1, x2, x3, w) != twostage_bf(x0, x1, x2, x3, w):
            fail("radix-4 butterfly numeric mismatch")


def check_forward_vs_golden(n):
    # impl forward NTT == independent direct-sum transform (up to the bit-reversal relabel)
    psi_n, zetas, _ = build_tables(n)
    rnd = random.Random(n)
    a = [rnd.randrange(Q) for _ in range(n)]
    g = golden_ntt(a, psi_n)
    o = ntt(a, zetas)
    logn = n.bit_length() - 1
    nat = [o[brv(k, logn)] for k in range(n)]
    if g != o and g != nat:
        fail("forward NTT mismatch vs golden at n=%d" % n)


def check_transform(n, trials):
    psi_n, zetas, zinv = build_tables(n)
    rnd = random.Random(1000 + n)
    for _ in range(trials):
        a = [rnd.randrange(Q) for _ in range(n)]
        if intt(ntt(a, zetas), zinv) != a:
            fail("INTT(NTT(a)) != a at n=%d" % n)
        b = [rnd.randrange(Q) for _ in range(n)]
        A = ntt(a, zetas)
        B = ntt(b, zetas)
        C = [mulmod(A[i], B[i]) for i in range(n)]
        c = intt(C, zinv)
        if c != negmul(a, b):
            fail("NTT-based negacyclic mult != schoolbook at n=%d" % n)


def deep_verify():
    # optional full-size forward NTT vs O(N^2) golden (slow); off by default
    psi_n, zetas, _ = build_tables(N)
    rnd = random.Random(99)
    a = [rnd.randrange(Q) for _ in range(N)]
    g = golden_ntt(a, psi_n)
    o = ntt(a, zetas)
    logn = N.bit_length() - 1
    nat = [o[brv(k, logn)] for k in range(N)]
    if g != o and g != nat:
        fail("DEEP: full-size forward NTT mismatch vs golden")


def main():
    check_constants()
    prove_barrett()
    prove_conflict_free()
    check_bijection()
    prove_radix4_fusion()
    check_radix4_numeric()
    # forward transform vs independent golden on small sizes (O(n^2) golden)
    check_forward_vs_golden(8)
    check_forward_vs_golden(16)
    # consistency + true correctness (round-trip + negacyclic mult)
    for n in (8, 16, 64, 256):
        check_transform(n, trials=3)
    # production size: structural correctness via fast round-trip / negacyclic mult
    check_transform(N, trials=1)
    if os.environ.get("DEEP_VERIFY") == "1":
        deep_verify()
    print("VERIFIED")
    sys.exit(0)


if __name__ == "__main__":
    main()
