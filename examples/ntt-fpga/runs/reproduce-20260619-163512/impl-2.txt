#!/usr/bin/env python3
"""
CFNTT Radix-2/4 NTT Multiplication Accelerator (FPGA) -- functional reproduction
from the scene descriptor SPEC ALONE.

Implements (the parts the spec actually pins down):
  * Barrett modular reduction (mu=21843, k=28) over 28-bit products.
  * The committed conflict-free XOR-fold bank/offset memory map.
  * The radix-2 Cooley-Tukey butterfly and the radix-4 fused butterfly (j=psi^512).
  * A negacyclic NTT/INTT (CT forward, GS inverse) over q=12289, N=1024, psi=1945.

Each correctness obligation is discharged with a SMALL, FINITE, SOUND check:
  * Barrett reduction        -> proved with z3 over the whole bounded input range.
  * Conflict-free bank map    -> exhaustively over the full N=1024 index space.
  * Butterfly == DFT golden   -> bounded exhaustive over a tiny prime field.
  * NTT/INTT round-trip + negacyclic-convolution identity -> exhaustive on a tiny
    field, randomized at production size (round-trip O(N log N); schoolbook conv
    at reduced sizes by default, full N=1024 conv only under DEEP_VERIFY=1).
  * FSM transition table      -> reachability over the finite state graph.

Prints "VERIFIED" / exits 0 on success; "FAIL: <reason+counterexample>" / exit 1.
"""

import os
import sys
import random

try:
    from z3 import Int, Solver, If, Or, unsat, sat
except ImportError:
    print("FAIL: z3 is required (pip install z3-solver) but is not importable")
    sys.exit(1)

random.seed(0xCF77)
DEEP = os.environ.get("DEEP_VERIFY") == "1"

# ----------------------------------------------------------------------------
# Authoritative parameters (verbatim from the spec's metadata.spec.params)
# ----------------------------------------------------------------------------
Q = 12289          # modulus
N = 1024           # transform length
PSI = 1945         # primitive 2N=2048-th root (negacyclic)
OMEGA = 10302      # psi^2, the N-th root
NINV = 12277       # 1024^-1 mod q
J = pow(PSI, 512, Q)   # constant radix-4 rotator, j^2 == -1
MU = 21843         # Barrett mu = floor(2^28 / q)
K = 28             # Barrett shift


# ----------------------------------------------------------------------------
# Datapath primitives
# ----------------------------------------------------------------------------
def barrett_reduce(x):
    """Reduce x in [0, Q^2) to [0, Q).  mod_red_mult (estimate) + mod_red_subshift
    (subtract & up-to-2 conditional corrections)."""
    qh = (x * MU) >> K          # mod_red_mult: qh = (x*mu) >> 28
    r = x - qh * Q              # mod_red_subshift: r = x - qh*q
    if r >= Q:                  # <= 2 conditional subtractions of q
        r -= Q
    if r >= Q:
        r -= Q
    return r


def mulmod_b(a, b):
    """Production modular multiply: 14b*14b -> 28b product -> Barrett-reduce."""
    return barrett_reduce(a * b)


def bank(i):
    """Committed conflict-free bank index: GF(2)-linear XOR-fold of the four
    3-bit groups of i."""
    return (i & 7) ^ ((i >> 3) & 7) ^ ((i >> 6) & 7) ^ ((i >> 9) & 7)


def offset(i):
    return i >> 3


def bf_radix2(x0, x1, w, q, mul):
    """Cooley-Tukey DIT radix-2 butterfly (config_radix_selector.radix2_butterfly):
    u=x0, v=w*x1 -> (u+v, u-v) mod q."""
    v = mul(w, x1)
    return (x0 + v) % q, (x0 - v) % q


def bf_radix4(x0, x1, x2, x3, w, q, mul, j):
    """Fused radix-4 butterfly (config_radix_selector.radix4_butterfly):
    b'=W1*x1, c'=W2*x2, d'=W3*x3 with W1=w,W2=w^2,W3=w^3; j the constant rotator.
    y0=x0+b'+c'+d'; y1=x0+j b'-c'-j d'; y2=x0-b'+c'-d'; y3=x0-j b'-c'+j d' (mod q)."""
    w2 = mul(w, w)
    w3 = mul(w2, w)
    b = mul(w, x1)
    c = mul(w2, x2)
    d = mul(w3, x3)
    jb = mul(j, b)
    jd = mul(j, d)
    y0 = (x0 + b + c + d) % q
    y1 = (x0 + jb - c - jd) % q
    y2 = (x0 - b + c - d) % q
    y3 = (x0 - jb - c + jd) % q
    return y0, y1, y2, y3


# ----------------------------------------------------------------------------
# Negacyclic NTT / INTT (Longa-Naehrig CT/GS pair) -- the polynomial engine
# ----------------------------------------------------------------------------
def _brv(x, bits):
    r = 0
    for _ in range(bits):
        r = (r << 1) | (x & 1)
        x >>= 1
    return r


def build_tables(q, psi, n):
    """Twiddle ROM contents: bit-reversed psi-power tables for the merged
    negacyclic transform, plus n^-1."""
    logn = n.bit_length() - 1
    psi_t = [pow(psi, _brv(i, logn), q) for i in range(n)]
    psii = pow(psi, q - 2, q)               # psi^-1 via Fermat
    psi_inv_t = [pow(psii, _brv(i, logn), q) for i in range(n)]
    ninv = pow(n, q - 2, q)
    return psi_t, psi_inv_t, ninv


def ntt(a, q, psi_t):
    """Forward negacyclic NTT (CT butterflies, bit-reversed output order)."""
    a = list(a)
    n = len(a)
    t = n
    m = 1
    while m < n:
        t //= 2
        for i in range(m):
            j1 = 2 * i * t
            s = psi_t[m + i]
            for j in range(j1, j1 + t):
                u = a[j]
                v = a[j + t] * s % q
                a[j] = (u + v) % q
                a[j + t] = (u - v) % q
        m *= 2
    return a


def intt(a, q, psi_inv_t, ninv):
    """Inverse negacyclic NTT (GS butterflies) + N^-1 scaling -> natural order."""
    a = list(a)
    n = len(a)
    t = 1
    m = n
    while m > 1:
        j1 = 0
        h = m // 2
        for i in range(h):
            s = psi_inv_t[h + i]
            for j in range(j1, j1 + t):
                u = a[j]
                v = a[j + t]
                a[j] = (u + v) % q
                a[j + t] = (u - v) * s % q
            j1 += 2 * t
        t *= 2
        m = h
    return [x * ninv % q for x in a]


def ntt_mul(a, b, q, psi_t, psi_inv_t, ninv):
    A = ntt(a, q, psi_t)
    B = ntt(b, q, psi_t)
    C = [A[i] * B[i] % q for i in range(len(a))]
    return intt(C, q, psi_inv_t, ninv)


def negconv_schoolbook(a, b, q):
    """Independent golden: schoolbook polynomial product mod (x^n + 1)."""
    n = len(a)
    c = [0] * n
    for i in range(n):
        ai = a[i]
        for jj in range(n):
            p = ai * b[jj] % q
            k = i + jj
            if k < n:
                c[k] = (c[k] + p) % q
            else:
                c[k - n] = (c[k - n] - p) % q
    return c


def dft4_golden(x, w, q, j):
    """Independent golden 4-point DFT of the twisted inputs:
    Y[k] = sum_m j^(m*k mod 4) * (w^m * x[m]) mod q."""
    tw = [1, w % q, (w * w) % q, (w * w * w) % q]
    t = [tw[m] * x[m] % q for m in range(4)]
    Y = []
    for k in range(4):
        s = 0
        for m in range(4):
            s = (s + pow(j, (m * k) % 4, q) * t[m]) % q
        Y.append(s)
    return tuple(Y)


# ----------------------------------------------------------------------------
# Checks
# ----------------------------------------------------------------------------
def check_params():
    if OMEGA != pow(PSI, 2, Q):
        return False, f"omega != psi^2: {OMEGA} vs {pow(PSI,2,Q)}"
    if pow(PSI, 1024, Q) != Q - 1:
        return False, "psi^1024 != -1 (not negacyclic)"
    if pow(PSI, 2048, Q) != 1:
        return False, "psi^2048 != 1"
    if (J * J) % Q != Q - 1:
        return False, f"j^2 != -1 (j={J})"
    if pow(11, 6, Q) != PSI:
        return False, "g^6 != psi (generator mismatch)"
    if MU != (1 << K) // Q:
        return False, f"mu != floor(2^28/q): {MU} vs {(1<<K)//Q}"
    if Q * Q >= (1 << K):
        return False, "q^2 not < 2^28 (Barrett bound broken)"
    if NINV != pow(N, Q - 2, Q) or (N * NINV) % Q != 1:
        return False, "Ninv != N^-1 mod q"
    return True, ""


def check_barrett_z3():
    """Prove: for all x in [0, q^2), barrett_reduce(x) == x mod q and in [0,q)."""
    x = Int('x')
    qh = Int('qh')      # qh = floor(x*MU / 2^K), modeled without z3 division
    rq = Int('rq')      # rq = floor(x / Q), the reference quotient
    s = Solver()
    s.add(x >= 0, x < Q * Q)
    SH = 1 << K
    s.add(qh >= 0, qh * SH <= x * MU, x * MU < (qh + 1) * SH)
    s.add(rq >= 0, rq * Q <= x, x < (rq + 1) * Q)
    r0 = x - qh * Q
    r1 = If(r0 >= Q, r0 - Q, r0)
    r2 = If(r1 >= Q, r1 - Q, r1)
    ref = x - rq * Q
    s.add(Or(r2 != ref, r2 < 0, r2 >= Q))   # negation: try to break it
    res = s.check()
    if res == unsat:
        return True, ""
    if res == sat:
        return False, f"Barrett counterexample x={s.model()[x].as_long()}"
    return False, "Barrett z3 returned unknown"


def check_bank_map():
    # (bank, offset) must be a bijection on [0, N).
    seen = {}
    for i in range(N):
        key = (bank(i), offset(i))
        if key in seen:
            return False, f"bank/offset collision: i={i} and i={seen[key]} -> {key}"
        seen[key] = i
    # GF(2)-linear => stride distinctness reduces to bank(2^s) != 0 for all strides.
    for sft in range(10):
        if bank(1 << sft) == 0:
            return False, f"stride 2^{sft} collides: bank(2^{sft})=0"
    # Explicit radix-2 pair distinctness across every stage/stride.
    for sft in range(10):
        st = 1 << sft
        for i in range(N):
            if (i & st) == 0 and i + st < N:
                if bank(i) == bank(i + st):
                    return False, f"r2 pair collide stride 2^{sft} at i={i}"
    # Radix-4 4-tuple (i, i+2^s, i+2^(s+1), i+3*2^s) must hit 4 distinct banks.
    for sft in range(9):
        st = 1 << sft
        for i in range(0, N - 3 * st):
            offs = (0, st, 2 * st, 3 * st)
            banks = {bank(i + o) for o in offs}
            if len(banks) != 4:
                return False, f"r4 tuple bank collide stride 2^{sft} at i={i}: {banks}"
    return True, ""


def check_butterflies_small_field():
    # Tiny prime field with the same algebraic structure: q=17, psi=2 (order 8),
    # so N=4, omega=4, j=4 (j^2 = 16 = -1).
    q, psi = 17, 2
    j = pow(psi, 2, q)          # 4th root of unity
    mul = lambda a, b: a * b % q
    # radix-2 butterfly == golden mod-q formula, exhaustive.
    for x0 in range(q):
        for x1 in range(q):
            for w in range(q):
                got = bf_radix2(x0, x1, w, q, mul)
                exp = ((x0 + w * x1) % q, (x0 - w * x1) % q)
                if got != exp:
                    return False, f"radix2 BF wrong x0={x0} x1={x1} w={w}: {got} != {exp}"
    # radix-4 butterfly == DFT-4 golden, bounded exhaustive.
    for x0 in range(9):
        for x1 in range(9):
            for x2 in range(9):
                for x3 in range(9):
                    for w in range(q):
                        got = bf_radix4(x0, x1, x2, x3, w, q, mul, j)
                        exp = dft4_golden([x0, x1, x2, x3], w, q, j)
                        if got != exp:
                            return False, (f"radix4 BF wrong x={[x0,x1,x2,x3]} w={w}: "
                                           f"{got} != {exp}")
    return True, ""


def check_ntt_small_field_exhaustive():
    # q=17, psi=2, N=4: prove round-trip identity over EVERY input vector.
    q, psi, n = 17, 2, 4
    psi_t, psi_inv_t, ninv = build_tables(q, psi, n)
    if pow(psi, 2 * n, q) != 1 or pow(psi, n, q) != q - 1:
        return False, "small-field psi has wrong order"
    for x0 in range(q):
        for x1 in range(q):
            for x2 in range(q):
                for x3 in range(q):
                    a = [x0, x1, x2, x3]
                    back = intt(ntt(a, q, psi_t), q, psi_inv_t, ninv)
                    if back != a:
                        return False, f"round-trip broke on {a} -> {back}"
    # negacyclic convolution identity vs schoolbook, random sample.
    for _ in range(400):
        a = [random.randrange(q) for _ in range(n)]
        b = [random.randrange(q) for _ in range(n)]
        if ntt_mul(a, b, q, psi_t, psi_inv_t, ninv) != negconv_schoolbook(a, b, q):
            return False, f"small-field conv mismatch a={a} b={b}"
    return True, ""


def check_ntt_production():
    # Derived roots stay inside q=12289: for length n | N, psi_n = PSI^(N/n)
    # has order 2n.  Verify round-trip + negacyclic conv at reduced sizes.
    for n in (16, 64, 256):
        psi_n = pow(PSI, N // n, Q)
        if pow(psi_n, 2 * n, Q) != 1 or pow(psi_n, n, Q) != Q - 1:
            return False, f"derived psi for n={n} has wrong order"
        psi_t, psi_inv_t, ninv = build_tables(Q, psi_n, n)
        for _ in range(8):
            a = [random.randrange(Q) for _ in range(n)]
            back = intt(ntt(a, Q, psi_t), Q, psi_inv_t, ninv)
            if back != a:
                return False, f"n={n} round-trip broke"
        for _ in range(4):
            a = [random.randrange(Q) for _ in range(n)]
            b = [random.randrange(Q) for _ in range(n)]
            if ntt_mul(a, b, Q, psi_t, psi_inv_t, ninv) != negconv_schoolbook(a, b, Q):
                return False, f"n={n} conv mismatch a={a[:4]}... b={b[:4]}..."

    # Full production size: tables must reproduce the spec's Ninv exactly.
    psi_t, psi_inv_t, ninv = build_tables(Q, PSI, N)
    if ninv != NINV:
        return False, f"production ninv {ninv} != spec {NINV}"
    for _ in range(5):
        a = [random.randrange(Q) for _ in range(N)]
        back = intt(ntt(a, Q, psi_t), Q, psi_inv_t, ninv)
        if back != a:
            return False, "N=1024 round-trip broke"
    if DEEP:
        a = [random.randrange(Q) for _ in range(N)]
        b = [random.randrange(Q) for _ in range(N)]
        if ntt_mul(a, b, Q, psi_t, psi_inv_t, ninv) != negconv_schoolbook(a, b, Q):
            return False, "N=1024 negacyclic conv mismatch"
    return True, ""


def check_production_butterflies_random():
    # The production butterflies use the z3-proven Barrett mulmod; differentially
    # test them against pure-bigint golden formulas at q=12289.
    j = pow(PSI, 512, Q)
    refmul = lambda a, b: a * b % Q
    for _ in range(5000):
        x0, x1, w = (random.randrange(Q) for _ in range(3))
        if bf_radix2(x0, x1, w, Q, mulmod_b) != bf_radix2(x0, x1, w, Q, refmul):
            return False, f"prod radix2 vs ref mismatch x0={x0} x1={x1} w={w}"
        x = [random.randrange(Q) for _ in range(4)]
        got = bf_radix4(*x, w, Q, mulmod_b, j)
        exp = dft4_golden(x, w, Q, j)
        if got != exp:
            return False, f"prod radix4 vs DFT4 mismatch x={x} w={w}: {got} != {exp}"
    return True, ""


def check_fsm():
    edges = [
        ("IDLE", "LOAD"), ("LOAD", "STAGE_LOOP"),
        ("STAGE_LOOP", "TWIDDLE_FETCH"), ("TWIDDLE_FETCH", "BUTTERFLY"),
        ("BUTTERFLY", "REDUCE"), ("REDUCE", "WRITEBACK"),
        ("WRITEBACK", "TWIDDLE_FETCH"), ("WRITEBACK", "STAGE_LOOP"),
        ("STAGE_LOOP", "INTT_STAGE_LOOP"), ("STAGE_LOOP", "DRAIN"),
        ("INTT_STAGE_LOOP", "SCALE_N_INV"), ("SCALE_N_INV", "DRAIN"),
        ("DRAIN", "IDLE"),
    ]
    adj = {}
    states = set()
    for a, b in edges:
        adj.setdefault(a, []).append(b)
        states.add(a)
        states.add(b)
    # BFS from IDLE; every declared state must be reachable.
    seen = {"IDLE"}
    stack = ["IDLE"]
    while stack:
        cur = stack.pop()
        for nxt in adj.get(cur, []):
            if nxt not in seen:
                seen.add(nxt)
                stack.append(nxt)
    if seen != states:
        return False, f"unreachable FSM states: {states - seen}"
    for terminal in ("DRAIN", "SCALE_N_INV"):
        if terminal not in seen:
            return False, f"{terminal} not reachable"
    return True, ""


def check_cycle_counts():
    # Spec's own stated identities (radix-2 path).
    stages_r2 = 10
    bf_per_stage_r2 = (N // 2) // 8           # 512 butterflies / 8 lanes = 64
    if bf_per_stage_r2 != 64:
        return False, f"r2 cycles/stage {bf_per_stage_r2} != 64"
    if stages_r2 * bf_per_stage_r2 != 640:
        return False, "r2 total cycles != 640"
    return True, ""


def main():
    checks = [
        ("authoritative parameters", check_params),
        ("Barrett reduction (z3, full range)", check_barrett_z3),
        ("conflict-free bank map (exhaustive)", check_bank_map),
        ("butterflies == DFT golden (small field)", check_butterflies_small_field),
        ("NTT round-trip + conv (small field exhaustive)", check_ntt_small_field_exhaustive),
        ("production butterflies (differential)", check_production_butterflies_random),
        ("NTT/INTT @ q=12289 (round-trip + conv)", check_ntt_production),
        ("schedule FSM reachability", check_fsm),
        ("radix-2 cycle accounting", check_cycle_counts),
    ]
    for name, fn in checks:
        ok, msg = fn()
        if not ok:
            print(f"FAIL: [{name}] {msg}")
            sys.exit(1)
    print("VERIFIED")
    sys.exit(0)


if __name__ == "__main__":
    main()
