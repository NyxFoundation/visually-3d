#!/usr/bin/env python3
"""
Self-checking MuJoCo build/sim of the CFNTT Radix-2/4 NTT Multiplication
Accelerator (FPGA), reconstructed FROM THE SPEC ALONE.

The real subject is an RTL accelerator with no moving parts, so the mechanical
model is an interpretation: static geoms reproduce the floorplan (substrate,
die, 8 BRAM banks, 8 butterfly lanes, Barrett reduction spine, twiddle ROM,
control). Three actuated DOFs make the spec's *dataflow* and *exploded views*
mechanical and checkable:
  - data_token : slide along the datapath X axis (crossbar -> reduction) = the
                 read-modify-write operand flow ("toolhead reaches build volume"
                 analog on the datapath axis).
  - exploded   : the lifted butterfly cluster riser (exploded view) slides up.
  - heatsink   : the exploded heatsink slides up.

Verifies L0 build, L1 stable physics, L2 actuated DOFs move, L3 the data_token
sweeps the full datapath span (both extremes) and the two exploded assemblies
reach their lifted positions.
"""

import sys
import math
import mujoco


# ---- spec coords are Y-up; MuJoCo is Z-up. Remap (x,y,z)_spec -> (x,z,y)_mj ---
def remap(p):
    x, y, z = p
    return (x, z, y)


def fmt(t):
    return " ".join(f"{v:.5f}" for v in t)


def sub(a, b):
    return (a[0] - b[0], a[1] - b[1], a[2] - b[2])


# ---- geom emitters -----------------------------------------------------------
def box_geom(size_spec, local_pos, rgba, density):
    sx, sy, sz = size_spec
    half = (sx / 2.0, sz / 2.0, sy / 2.0)  # remap full-extents -> mj half-extents
    return (f'<geom type="box" size="{fmt(half)}" pos="{fmt(local_pos)}" '
            f'rgba="{rgba}" density="{density}" contype="0" conaffinity="0"/>')


def cyl_geom(size_spec, local_pos, rgba, density):
    r, h = size_spec[0], size_spec[1]
    return (f'<geom type="cylinder" size="{r:.5f} {h/2.0:.5f}" '
            f'pos="{fmt(local_pos)}" rgba="{rgba}" density="{density}" '
            f'contype="0" conaffinity="0"/>')


# ---- part tables (id, kind, spec-pos, spec-size) -----------------------------
GREY = "0.55 0.57 0.60 1"
GLASS = "0.40 0.65 0.85 0.5"
COPPER = "0.80 0.45 0.20 1"
DARK = "0.20 0.20 0.22 1"
BRASS = "0.80 0.65 0.25 1"
STEEL = "0.70 0.72 0.75 1"
WHITE = "0.90 0.90 0.92 1"

LANE_Z = [-3.85, -2.75, -1.65, -0.55, 0.55, 1.65, 2.75, 3.85]

static = []  # (kind, spec_pos, spec_size, rgba)

# board + die
static.append(("box", [0, 0.2, 0], [14, 0.4, 9], DARK))
static.append(("box", [0, 0.46, 0], [12, 0.08, 7.4], STEEL))
# 8 interleaved BRAM banks
for z in LANE_Z:
    static.append(("box", [-5.8, 1, z], [1.3, 1.2, 0.85], GLASS))
# FIFOs (capsules -> approximated as bounding boxes; rotation is decoration)
static.append(("box", [-6.8, 0.95, 3.2], [0.9, 0.9, 1.8], WHITE))
static.append(("box", [-6.8, 0.95, -3.2], [0.9, 0.9, 1.8], WHITE))
# crossbar, intt scale, butterfly baseplate
static.append(("box", [-4.2, 0.9, 0], [1.2, 1.0, 8.4], DARK))
static.append(("box", [-3.25, 0.7, 3], [0.85, 0.6, 1.0], BRASS))
static.append(("box", [0, 0.475, 0], [3.6, 0.15, 8.4], STEEL))
# 8 butterfly lanes: DSP mult (cyl), adder, subtractor, pipeline reg
for z in LANE_Z:
    static.append(("cyl", [-1.1, 0.95, z], [0.35, 0.9], COPPER))
    static.append(("box", [0.5, 0.72, z - 0.24], [1.1, 0.66, 0.42], DARK))
    static.append(("box", [0.5, 0.72, z + 0.24], [1.1, 0.66, 0.42], STEEL))
    static.append(("box", [1.5, 0.68, z], [0.4, 0.52, 0.8], WHITE))
# twiddle rail + feeder, read bus
static.append(("box", [-1.1, 1.65, 0], [0.28, 0.12, 8.0], BRASS))
static.append(("box", [1.2, 1.65, -2.5], [4.6, 0.12, 0.28], BRASS))
static.append(("box", [-2.65, 1.45, 0], [3.3, 0.1, 0.32], COPPER))
# Barrett reduction spine (rotated cyl -> bounding box) + shift/sub member
static.append(("box", [2.2, 0.95, 0], [1.0, 1.0, 8.0], GREY))
static.append(("box", [3.1, 0.85, 0], [0.7, 0.9, 8.0], DARK))
# return bus
static.append(("box", [-0.55, 1.62, -4], [7.3, 0.1, 0.35], GLASS))
# twiddle ROM, addr-gen, config, schedule, PLL
static.append(("box", [4.6, 1.15, -2.5], [2.2, 1.5, 2.6], GLASS))
static.append(("box", [4.6, 0.95, 2.5], [2.2, 1.1, 2.6], DARK))
static.append(("box", [6.1, 0.7, 1.2], [0.9, 0.6, 1.1], DARK))
static.append(("box", [6.1, 0.85, 3.4], [1.0, 0.9, 1.3], GREY))
static.append(("cyl", [6.2, 0.8, -1.2], [0.5, 0.8], STEEL))

# ---- moving assemblies -------------------------------------------------------
# exploded butterfly cluster (lifted on risers)
EX_ORIGIN = [0, 2.3, 2.4]
ex_parts = [
    ("box", [0, 2.3, 2.4], [3.6, 0.12, 1.6], STEEL),
    ("cyl", [-1.1, 2.75, 2.4], [0.4, 0.9], COPPER),
    ("box", [0.5, 2.65, 2.16], [1.1, 0.65, 0.42], DARK),
    ("box", [0.5, 2.65, 2.64], [1.1, 0.65, 0.42], STEEL),
    ("box", [1.5, 2.6, 2.4], [0.4, 0.55, 0.9], WHITE),
    ("cyl", [-1.5, 1.39, 2.4], [0.07, 1.7], STEEL),
    ("cyl", [1.5, 1.39, 2.4], [0.07, 1.7], STEEL),
]
# exploded heatsink (lifted on a riser stub)
HS_ORIGIN = [4.6, 2.75, 0]
hs_parts = [("box", [4.6, 2.75, 0], [3.0, 0.2, 4.0], STEEL),
            ("cyl", [4.6, 1.6, 0], [0.08, 2.2], STEEL)]
for i in range(7):
    hs_parts.append(("box", [3.25 + 0.45 * i, 3.25, 0], [0.18, 0.85, 3.7], STEEL))


def emit(kind, spec_pos, spec_size, rgba, origin, density):
    local = sub(remap(spec_pos), remap(origin))
    if kind == "box":
        return box_geom(spec_size, local, rgba, density)
    return cyl_geom(spec_size, local, rgba, density)


def build_mjcf():
    sgeoms = "\n      ".join(
        emit(k, p, s, c, (0, 0, 0), 200) for (k, p, s, c) in static)
    ex_o = remap(EX_ORIGIN)
    hs_o = remap(HS_ORIGIN)
    ex_geoms = "\n        ".join(
        emit(k, p, s, c, EX_ORIGIN, 50) for (k, p, s, c) in ex_parts)
    hs_geoms = "\n        ".join(
        emit(k, p, s, c, HS_ORIGIN, 50) for (k, p, s, c) in hs_parts)
    # data token starts at the crossbar/datapath center, slides along world X
    tok_o = remap([-0.5, 1.45, 0])
    return f"""<mujoco model="cfntt_accelerator">
  <compiler inertiafromgeom="true" angle="radian"/>
  <option gravity="0 0 -9.81" timestep="0.002"/>
  <worldbody>
    <geom name="floor" type="plane" size="40 40 0.1" rgba="0.3 0.3 0.33 1"/>
      {sgeoms}

    <body name="data_token" pos="{fmt(tok_o)}">
      <joint name="j_token" type="slide" axis="1 0 0" range="-3 3" damping="20"/>
      <geom type="box" size="0.18 0.18 0.18" rgba="1 0.85 0.1 1"
            density="30" contype="0" conaffinity="0"/>
    </body>

    <body name="exploded" pos="{fmt(ex_o)}">
      <joint name="j_exploded" type="slide" axis="0 0 1" range="0 1.0" damping="60"/>
        {ex_geoms}
    </body>

    <body name="heatsink" pos="{fmt(hs_o)}">
      <joint name="j_heatsink" type="slide" axis="0 0 1" range="0 1.2" damping="60"/>
        {hs_geoms}
    </body>
  </worldbody>

  <actuator>
    <position name="a_token"    joint="j_token"    kp="4000" ctrlrange="-3 3"/>
    <position name="a_exploded" joint="j_exploded" kp="900000" ctrlrange="0 1.0"/>
    <position name="a_heatsink" joint="j_heatsink" kp="900000" ctrlrange="0 1.2"/>
  </actuator>
</mujoco>"""


def finite(arr):
    import numpy as np
    return np.all(np.isfinite(arr))


def main():
    import numpy as np

    MJCF = build_mjcf()

    # ----- L0: builds -----
    try:
        model = mujoco.MjModel.from_xml_string(MJCF)
        data = mujoco.MjData(model)
    except Exception as e:  # noqa: BLE001
        print(f"L0 FAIL: model did not build: {e}")
        sys.exit(1)
    nbody = model.nbody
    nu = model.nu

    # ----- L1: valid physics under gravity (no actuation) -----
    mujoco.mj_resetData(model, data)
    for _ in range(1000):
        mujoco.mj_step(model, data)
        if not (finite(data.qpos) and finite(data.qvel)):
            print("L1 FAIL: NaN/Inf in state")
            sys.exit(1)
    if np.max(np.abs(data.qpos)) > 50.0 or np.max(np.abs(data.qvel)) > 200.0:
        print("L1 FAIL: model exploded (unbounded state)")
        sys.exit(1)

    # ----- L2: each actuator measurably moves its joint -----
    jnt_adr = {}
    for a in range(nu):
        jid = model.actuator_trnid[a, 0]
        jnt_adr[a] = model.jnt_qposadr[jid]
    targets = {0: 2.5, 1: 0.8, 2: 0.9}   # token, exploded, heatsink
    thresh = {0: 0.3, 1: 0.05, 2: 0.05}
    for a in range(nu):
        mujoco.mj_resetData(model, data)
        for _ in range(50):
            mujoco.mj_step(model, data)
        q0 = data.qpos[jnt_adr[a]]
        data.ctrl[:] = 0.0
        data.ctrl[a] = targets[a]
        for _ in range(600):
            mujoco.mj_step(model, data)
        dq = abs(data.qpos[jnt_adr[a]] - q0)
        if dq < thresh[a]:
            print(f"L2 FAIL: actuator {a} barely moved (|dq|={dq:.4f})")
            sys.exit(1)

    # ----- L3: datapath token sweeps full span; exploded assemblies lift -----
    def settle(ctrl, steps=700):
        for k, v in ctrl.items():
            data.ctrl[k] = v
        for _ in range(steps):
            mujoco.mj_step(model, data)

    tok_bid = model.body("data_token").id
    ex_bid = model.body("exploded").id
    hs_bid = model.body("heatsink").id

    mujoco.mj_resetData(model, data)
    # drive token to the reduction (max X) end of the read-modify-write loop
    settle({0: 3.0, 1: 0.8, 2: 0.9})
    x_max = data.xpos[tok_bid, 0]
    ex_z = data.xpos[ex_bid, 2]
    hs_z = data.xpos[hs_bid, 2]
    # drive token back to the crossbar (min X) end
    settle({0: -3.0, 1: 0.8, 2: 0.9})
    x_min = data.xpos[tok_bid, 0]

    span = x_max - x_min
    ex_lift = ex_z - remap(EX_ORIGIN)[2]
    hs_lift = hs_z - remap(HS_ORIGIN)[2]

    if span < 5.0:
        print(f"L3 FAIL: datapath token span too small ({span:.3f} m)")
        sys.exit(1)
    if ex_lift < 0.4:
        print(f"L3 FAIL: exploded butterfly cluster did not lift ({ex_lift:.3f} m)")
        sys.exit(1)
    if hs_lift < 0.4:
        print(f"L3 FAIL: exploded heatsink did not lift ({hs_lift:.3f} m)")
        sys.exit(1)

    print("VERIFIED")
    print(f"L0 build: {nbody} bodies, {nu} actuators, valid kinematic tree | "
          f"L1 stable: 1000 steps, |qpos|<50 finite | "
          f"L2 DOF: 3/3 actuators move their joints | "
          f"L3 task: datapath span={span:.2f}m (x[{x_min:.2f},{x_max:.2f}]), "
          f"exploded lift={ex_lift:.2f}m, heatsink lift={hs_lift:.2f}m")
    sys.exit(0)


if __name__ == "__main__":
    main()
