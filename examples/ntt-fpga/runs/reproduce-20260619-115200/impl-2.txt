#!/usr/bin/env python3
"""
CFNTT Radix-2/4 NTT Multiplication Accelerator (FPGA) -- functional reproduction
from spec alone.  Single self-contained, self-checking script.

We implement ONLY the functional datapath the spec pins down:
  * the field constants (q, psi, omega, Ninv, j, Barrett mu/k),
  * the negacyclic NTT / INTT (radix-2 CT/GS, the datapath the 8 lanes run),
  * the radix-2 and radix-4 butterfly algebra,
  * the conflict-free interleaved-bank address map.

Verification turns correctness into a few SMALL FINITE obligations:
  1. constants satisfy the negacyclic root / inverse / Barrett relations (concrete);
  2. Barrett reduction is correct on its whole legal input range [0, q^2)
     -- proved with z3 over a LINEAR encoding (qh as a bounded variable);
  3. the radix-4 butterfly == two radix-2 stages -- proved with z3 (ring identity)
     and re-checked exhaustively over a small field;
  4. the conflict-free bank map is a bijection AND gives distinct banks for every
     radix-2 / radix-4 operand group at every stage -- exhaustive over [0,N);
  5. NTT/INTT is the identity, and NTT-based pointwise multiply equals schoolbook
     negacyclic convolution -- exhaustive-ish on small N, round-trip on N=1024.

Prints "VERIFIED" / exit 0 on success, or "FAIL: ..." / exit 1.
Deterministic, finishes well under 60s.  Heavy O(N^2) full-size convolution is
behind DEEP_VERIFY=1 only.
"""

import os
import sys
import random

try:
    from z3 import Int, Solver, Or, unsat
except Exception as e:  # z3 is an allowed dependency; without it we cannot prove
    print("FAIL: z3 is required but not importable: %r" % (e,))
    sys.exit(1)

# ----------------------------------------------------------------------------
# Authoritative parameters (metadata.spec.params)
# ----------------------------------------------------------------------------
Q        = 12289          # 14-bit NTT-friendly prime, q = 3*2^12 + 1
N        = 1024           # transform length
PSI      = 1945           # primitive 2N-th root (negacyclic), order 2048
OMEGA    = 10302          # psi^2, N-th root
NINV     = 12277          # 1024^-1 mod q
GEN_G    = 11             # generator, psi = g^6
BAR_MU   = 21843          # floor(2^28 / q)
BAR_K    = 28
NLANES   = 8              # parallel butterfly units
COEF_W   = 14             # coefficient width
PROD_W   = 28             # product width


def fail(msg):
    print("FAIL: " + msg)
    sys.exit(1)


# ----------------------------------------------------------------------------
# (1) Concrete constant relations
# ----------------------------------------------------------------------------
def check_constants():
    if pow(GEN_G, 6, Q) != PSI:
        fail("psi != g^6 mod q: g^6=%d, psi=%d" % (pow(GEN_G, 6, Q), PSI))
    if pow(PSI, 2, Q) != OMEGA:
        fail("omega != psi^2: got %d, want %d" % (pow(PSI, 2, Q), OMEGA))
    if pow(PSI, N, Q) != Q - 1:
        fail("psi^N != -1 (not negacyclic): %d" % pow(PSI, N, Q))
    if pow(PSI, 2 * N, Q) != 1:
        fail("psi^(2N) != 1: %d" % pow(PSI, 2 * N, Q))
    if pow(OMEGA, N, Q) != 1 or pow(OMEGA, N // 2, Q) == 1:
        fail("omega is not a primitive N-th root")
    if pow(N, -1, Q) != NINV:
        fail("Ninv wrong: pow(N,-1,q)=%d, spec=%d" % (pow(N, -1, Q), NINV))
    j = pow(PSI, N // 2, Q)              # psi^512, the radix-4 constant rotator
    if (j * j) % Q != Q - 1:
        fail("j=psi^512 is not a 4th root of unity (j^2 != -1): j=%d" % j)
    if (2 ** BAR_K) // Q != BAR_MU:
        fail("barrett mu wrong: floor(2^28/q)=%d, spec=%d"
             % ((2 ** BAR_K) // Q, BAR_MU))
    if Q * Q >= 2 ** BAR_K:
        fail("q^2 not < 2^28; Barrett input bound broken")
    return j


# ----------------------------------------------------------------------------
# (2) Barrett modular reduction (mod_red_mult + mod_red_subshift)
# ----------------------------------------------------------------------------
def barrett_reduce(x):
    # x in [0, q^2)
    qh = (x * BAR_MU) >> BAR_K          # mod_red_mult: quotient estimate
    r = x - qh * Q                      # mod_red_subshift: r = x - qh*q
    while r >= Q:                       # <=2 conditional subtractions
        r -= Q
    return r


def modmul(a, b):
    return barrett_reduce(a * b)


def prove_barrett():
    """Prove: for all x in [0, q^2), with qh = floor(x*mu/2^28),
    r = x - qh*q satisfies 0 <= r < 3q.  r == x (mod q) holds by construction
    (qh*q is a multiple of q), so <=2 conditional subtractions land r in [0,q).
    Encoding qh as a bounded Int variable keeps this LINEAR -> fast & decidable."""
    x = Int('x')
    qh = Int('qh')
    s = Solver()
    s.add(x >= 0, x < Q * Q)
    twoK = 2 ** BAR_K
    # qh = floor(x*mu/2^28) pinned by the two enclosing inequalities
    s.add(qh * twoK <= x * BAR_MU, x * BAR_MU < (qh + 1) * twoK)
    r = x - qh * Q
    s.add(Or(r < 0, r >= 3 * Q))        # negation of the claim
    res = s.check()
    if res != unsat:
        m = s.model()
        fail("Barrett bound violated, counterexample x=%s" % m.eval(x))


def check_barrett_concrete():
    rnd = random.Random(1)
    # boundary + random products of two reduced coefficients
    samples = [0, 1, Q - 1, Q, Q * Q - 1]
    for _ in range(20000):
        a = rnd.randrange(Q)
        b = rnd.randrange(Q)
        samples.append(a * b)
    for x in samples:
        got = barrett_reduce(x)
        if got != x % Q or not (0 <= got < Q):
            fail("barrett_reduce(%d)=%d, want %d" % (x, got, x % Q))


# ----------------------------------------------------------------------------
# (3) Radix-2 and radix-4 butterfly algebra
# ----------------------------------------------------------------------------
def bf_radix2(x0, x1, w):
    """CT butterfly: u=x0, v=w*x1 -> (u+v, u-v) mod q."""
    u = x0
    v = modmul(w, x1)
    return (u + v) % Q, (u - v) % Q


def bf_radix4(x0, x1, x2, x3, w, j):
    """Fused radix-4 BU with W1=w, W2=w^2, W3=w^3 and constant rotator j."""
    w2 = modmul(w, w)
    w3 = modmul(w2, w)
    b = modmul(w, x1)
    c = modmul(w2, x2)
    d = modmul(w3, x3)
    y0 = (x0 + b + c + d) % Q
    y1 = (x0 + modmul(j, b) - c - modmul(j, d)) % Q
    y2 = (x0 - b + c - d) % Q
    y3 = (x0 - modmul(j, b) - c + modmul(j, d)) % Q
    return y0, y1, y2, y3


def prove_radix4_is_two_radix2():
    """Ring identity: the spec's 4-output add/sub tree equals composing two
    radix-2 butterflies on {x0+c, x0-c} and {b+d, b-d} with rotator j (j^2=-1).
    Work at the (b',c',d') level so this is a clean linear identity in z3."""
    x0, b, c, d, j = (Int('x0'), Int('b'), Int('c'), Int('d'), Int('j'))
    s = Solver()
    s.add(j * j == -1)
    # implementation (mod-free ring form of bf_radix4's tree)
    y0 = x0 + b + c + d
    y1 = x0 + j * b - c - j * d
    y2 = x0 - b + c - d
    y3 = x0 - j * b - c + j * d
    # independent golden: two radix-2 stages
    A0, A1 = x0 + c, x0 - c
    B0, B1 = b + d, b - d
    g0, g2 = A0 + B0, A0 - B0
    g1, g3 = A1 + j * B1, A1 - j * B1
    s.add(Or(y0 != g0, y1 != g1, y2 != g2, y3 != g3))
    if s.check() != unsat:
        fail("radix-4 butterfly is NOT equivalent to two radix-2 stages")
    # exhaustive cross-check over a small field with a real 4th root (mod 13, j=5)
    p, jj = 13, 5
    assert (jj * jj) % p == p - 1
    for x0v in range(p):
        for bv in range(p):
            for cv in range(p):
                for dv in range(p):
                    yy0 = (x0v + bv + cv + dv) % p
                    yy1 = (x0v + jj * bv - cv - jj * dv) % p
                    yy2 = (x0v - bv + cv - dv) % p
                    yy3 = (x0v - jj * bv - cv + jj * dv) % p
                    a0, a1 = (x0v + cv) % p, (x0v - cv) % p
                    bb0, bb1 = (bv + dv) % p, (bv - dv) % p
                    if (yy0, yy1, yy2, yy3) != (
                        (a0 + bb0) % p, (a1 + jj * bb1) % p,
                        (a0 - bb0) % p, (a1 - jj * bb1) % p,
                    ):
                        fail("radix-4 vs radix-2 mismatch over GF(13)")


# ----------------------------------------------------------------------------
# (4) Conflict-free interleaved-bank address map (crossbar_network.spec)
# ----------------------------------------------------------------------------
def bank(i):
    return (i & 7) ^ ((i >> 3) & 7) ^ ((i >> 6) & 7) ^ ((i >> 9) & 7)


def offset(i):
    return i >> 3


def check_conflict_free():
    # (a) (bank, offset) is a bijection on [0, N)
    seen = set()
    for i in range(N):
        key = (bank(i), offset(i))
        if key in seen:
            fail("bank/offset collision: index %d maps to %r" % (i, key))
        seen.add(key)
    if len(seen) != N:
        fail("bank/offset not a bijection: %d distinct slots" % len(seen))

    log2n = N.bit_length() - 1            # 10

    # (b) radix-2: every stage stride 2^s pairs (i, i|2^s) into distinct banks.
    #     The DIT low index has bit s clear, so the two differ in exactly bit s.
    for s in range(log2n):
        t = 1 << s
        for i in range(N):
            if i & t:                     # only the low partner of each pair
                continue
            if bank(i) == bank(i | t):
                fail("radix-2 bank conflict at stage s=%d, i=%d" % (s, i))

    # (c) radix-4: fuse stages (s, s+1); operands are the 4 combos of bits s,s+1.
    #     Schedule fuses (1,0),(3,2),(5,4),(7,6),(9,8) -> low stage s in {0,2,4,6,8}.
    for s in range(0, log2n - 1, 2):
        bs, bs1 = 1 << s, 1 << (s + 1)
        mask = bs | bs1
        for i in range(N):
            if i & mask:                  # i must have both fused bits clear
                continue
            grp = (i, i | bs, i | bs1, i | bs | bs1)
            banks = {bank(g) for g in grp}
            if len(banks) != 4:
                fail("radix-4 bank conflict at fused stage s=%d, i=%d: banks=%r"
                     % (s, i, sorted(banks)))


# ----------------------------------------------------------------------------
# (5) Negacyclic NTT / INTT datapath (the 8 lanes run this; merged psi twist)
# ----------------------------------------------------------------------------
def bitrev(x, bits):
    r = 0
    for _ in range(bits):
        r = (r << 1) | (x & 1)
        x >>= 1
    return r


def psi_table(n, base):
    """psi-power table in bit-reversed order: tbl[k] = base^(bitrev(k))."""
    bits = n.bit_length() - 1
    pw = [pow(base, i, Q) for i in range(n)]
    return [pw[bitrev(k, bits)] for k in range(n)]


def ntt_forward(a, tbl):
    """In-place merged-negacyclic forward NTT (Cooley-Tukey, DIT)."""
    a = list(a)
    n = len(a)
    t = n
    m = 1
    while m < n:
        t //= 2
        for i in range(m):
            j1 = 2 * i * t
            S = tbl[m + i]
            for jx in range(j1, j1 + t):
                U = a[jx]
                V = modmul(S, a[jx + t])
                a[jx] = (U + V) % Q
                a[jx + t] = (U - V) % Q
        m *= 2
    return a


def intt_inverse(a, tbl_inv, ninv):
    """In-place merged-negacyclic inverse NTT (Gentleman-Sande)."""
    a = list(a)
    n = len(a)
    t = 1
    m = n
    while m > 1:
        j1 = 0
        h = m // 2
        for i in range(h):
            S = tbl_inv[h + i]
            for jx in range(j1, j1 + t):
                U = a[jx]
                V = a[jx + t]
                a[jx] = (U + V) % Q
                a[jx + t] = modmul((U - V) % Q, S)
            j1 += 2 * t
        t *= 2
        m = h
    return [modmul(x, ninv) for x in a]


def neg_conv_schoolbook(a, b):
    """Golden negacyclic convolution mod (x^n + 1), independent of the NTT."""
    n = len(a)
    c = [0] * n
    for i in range(n):
        ai = a[i]
        for j in range(n):
            k = i + j
            if k < n:
                c[k] = (c[k] + ai * b[j]) % Q
            else:
                c[k - n] = (c[k - n] - ai * b[j]) % Q
    return [x % Q for x in c]


def check_ntt():
    rnd = random.Random(7)

    # small-N: full equivalence vs schoolbook negacyclic convolution + round-trip
    for n in (2, 4, 8, 16):
        psi_n = pow(GEN_G, (2 * 6) * (N // (2 * n)) // 1, Q) if False else None
        # Derive an order-2n primitive root for this n from the order-2N psi:
        psi_n = pow(PSI, N // n, Q)        # has order 2n; psi_n^n == -1
        if pow(psi_n, n, Q) != Q - 1:
            fail("derived psi for n=%d is not negacyclic" % n)
        ninv_n = pow(n, -1, Q)
        psi_inv = pow(psi_n, -1, Q)
        tbl = psi_table(n, psi_n)
        tbl_inv = psi_table(n, psi_inv)

        for _ in range(40):
            a = [rnd.randrange(Q) for _ in range(n)]
            b = [rnd.randrange(Q) for _ in range(n)]
            # round-trip identity: INTT(NTT(a)) == a
            rt = intt_inverse(ntt_forward(a, tbl), tbl_inv, ninv_n)
            if rt != a:
                fail("round-trip identity failed at n=%d: %r != %r" % (n, rt, a))
            # convolution: INTT(NTT(a) o NTT(b)) == schoolbook negacyclic conv
            fa = ntt_forward(a, tbl)
            fb = ntt_forward(b, tbl)
            fp = [modmul(x, y) for x, y in zip(fa, fb)]
            got = intt_inverse(fp, tbl_inv, ninv_n)
            want = neg_conv_schoolbook(a, b)
            if got != want:
                fail("negacyclic conv mismatch at n=%d:\n got=%r\nwant=%r"
                     % (n, got, want))

    # production N=1024: round-trip identity (O(N log N), cheap), several vectors
    tbl = psi_table(N, PSI)
    tbl_inv = psi_table(N, pow(PSI, -1, Q))
    for _ in range(5):
        a = [rnd.randrange(Q) for _ in range(N)]
        rt = intt_inverse(ntt_forward(a, tbl), tbl_inv, NINV)
        if rt != a:
            fail("N=1024 round-trip identity failed")

    # production-size convolution vs schoolbook is O(N^2): opt-in only.
    if os.environ.get("DEEP_VERIFY") == "1":
        a = [rnd.randrange(Q) for _ in range(N)]
        b = [rnd.randrange(Q) for _ in range(N)]
        fp = [modmul(x, y)
              for x, y in zip(ntt_forward(a, tbl), ntt_forward(b, tbl))]
        got = intt_inverse(fp, tbl_inv, NINV)
        if got != neg_conv_schoolbook(a, b):
            fail("N=1024 negacyclic convolution mismatch (DEEP_VERIFY)")


# ----------------------------------------------------------------------------
# Radix-2 datapath sanity: an NTT built from bf_radix2 == the merged engine on
# a single stage (the bf_radix2 op equals the engine's inner update).
# ----------------------------------------------------------------------------
def check_radix2_butterfly():
    rnd = random.Random(99)
    for _ in range(5000):
        x0 = rnd.randrange(Q)
        x1 = rnd.randrange(Q)
        w = rnd.randrange(Q)
        y0, y1 = bf_radix2(x0, x1, w)
        ref0 = (x0 + w * x1) % Q
        ref1 = (x0 - w * x1) % Q
        if (y0, y1) != (ref0, ref1):
            fail("radix-2 butterfly mismatch: x0=%d x1=%d w=%d" % (x0, x1, w))


def check_radix4_butterfly_concrete():
    j = pow(PSI, N // 2, Q)
    rnd = random.Random(123)
    for _ in range(5000):
        xs = [rnd.randrange(Q) for _ in range(4)]
        w = rnd.randrange(Q)
        y = bf_radix4(xs[0], xs[1], xs[2], xs[3], w, j)
        w2 = (w * w) % Q
        w3 = (w2 * w) % Q
        b = (w * xs[1]) % Q
        c = (w2 * xs[2]) % Q
        d = (w3 * xs[3]) % Q
        ref = (
            (xs[0] + b + c + d) % Q,
            (xs[0] + j * b - c - j * d) % Q,
            (xs[0] - b + c - d) % Q,
            (xs[0] - j * b - c + j * d) % Q,
        )
        if y != ref:
            fail("radix-4 butterfly modmul mismatch")


# ----------------------------------------------------------------------------
def main():
    j = check_constants()
    _ = j
    prove_barrett()
    check_barrett_concrete()
    check_radix2_butterfly()
    prove_radix4_is_two_radix2()
    check_radix4_butterfly_concrete()
    check_conflict_free()
    check_ntt()
    print("VERIFIED")
    sys.exit(0)


if __name__ == "__main__":
    main()
