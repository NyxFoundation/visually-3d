#!/usr/bin/env python3
"""
CFNTT Radix-2/4 NTT Multiplication Accelerator (FPGA) — reverse-implemented from
spec ALONE. Functional Python model + two-tier self-check.

Authoritative params taken verbatim from metadata.spec (NOT counted as guesses):
  N=1024, q=12289, psi=1945 (order 2048, negacyclic), omega=psi^2=10302,
  Ninv=12277, barrett mu=21843, k=28, 8 banks/BU, bank XOR-fold + offset,
  radix-4 BU = 2 mults + 8 add/sub, e2e_N=16.

Everything the spec did NOT pin (butterfly algorithm, ROM addressing, sigma_s,
FSM walk, host DMA framing) is a reproduction choice — see PART-1 metadata.

Tier 1: size-INDEPENDENT facts proven at full bit-width (z3 + finite/exhaustive):
  - Barrett reduction correct & in [0,q) for ALL x in [0,(q-1)^2]
  - root constraints (psi^1024=-1, omega=psi^2, ninv, j^2=-1)
  - bank() conflict-free for every butterfly pair (i,i+2^s) at every stride
  - (bank,offset) is a bijection on [0,N)
  - radix-2 / radix-4 butterfly ops mod q, radix-4 fusion identity (= DFT4)
  - radix-4 op-count reductions (33%/20% vs naive 3/10)
  - host DMA 64b beat pack/unpack round-trip
Tier 2: whole-system equivalence at the SMALLEST structure-preserving size
  (N=16, same prime q=12289) against an INDEPENDENT schoolbook golden:
  - GS_INTT(CT_NTT(x)) == x  (round trip)
  - negacyclic poly mult via the NTT datapath == schoolbook reduction mod x^N+1
"""

import os
import random
from z3 import BitVec, LShR, If, UGE, ULT, ULE, URem, Solver, Not, And, sat

# ----------------------------------------------------------------------------
# Authoritative parameters (metadata.spec) — production instance
# ----------------------------------------------------------------------------
Q          = 12289
N_PROD     = 1024
BANKS      = 8
PSI        = 1945          # primitive 2048-th root, negacyclic
OMEGA      = 10302         # psi^2
NINV_PROD  = 12277         # 1024^-1 mod q
GEN_G      = 11
BARRETT_MU = 21843         # floor(2^28 / q)
BARRETT_K  = 28
COEFF_BITS = 14
PROD_BITS  = 28
E2E_N      = 16            # metadata.spec.verification.e2e_N

# radix-4 op counts (metadata)
R4_MULT_NAIVE, R4_MULT_OPT     = 3, 2
R4_ADDSUB_NAIVE, R4_ADDSUB_OPT = 10, 8


def fail(msg: str) -> None:
    print("FAIL: " + msg)
    raise SystemExit(1)


# ----------------------------------------------------------------------------
# Modular reduction — Barrett (mod_red_mult + mod_red_subshift)
# ----------------------------------------------------------------------------
def barrett_reduce(x: int, q: int = Q, mu: int = BARRETT_MU, k: int = BARRETT_K) -> int:
    """t=(x*mu)>>k ; r=x-t*q ; <=2 conditional subtractions -> r in [0,q).
    Valid for x in [0,(q-1)^2]."""
    t = (x * mu) >> k
    r = x - t * q
    while r >= q:          # spec: at most 2 iterations for x in [0,(q-1)^2]
        r -= q
    return r


def mulmod(a: int, b: int, q: int = Q) -> int:
    return barrett_reduce((a % q) * (b % q), q)


# ----------------------------------------------------------------------------
# Conflict-free memory mapping (addr_gen_unit / crossbar_network) — committed
# XOR-fold; offset = i>>3.  [source-missing: paper's Sec.III function]
# ----------------------------------------------------------------------------
def bank_fn(i: int) -> int:
    return (i & 7) ^ ((i >> 3) & 7) ^ ((i >> 6) & 7) ^ ((i >> 9) & 1)


def offset_fn(i: int) -> int:
    return i >> 3


# ----------------------------------------------------------------------------
# Twiddle ROM — bit-reversed psi-power table (CT/GS layout). [conv]
# ----------------------------------------------------------------------------
def bitrev(x: int, bits: int) -> int:
    r = 0
    for _ in range(bits):
        r = (r << 1) | (x & 1)
        x >>= 1
    return r


def psi_rev_table(psi: int, n: int, q: int) -> list:
    bits = n.bit_length() - 1
    return [pow(psi, bitrev(i, bits), q) for i in range(n)]


# ----------------------------------------------------------------------------
# Butterfly array — radix-2 (CT forward / GS inverse) and radix-4 fusion
# ----------------------------------------------------------------------------
def CT_NTT(a, psirev, q):
    """Cooley-Tukey DIT, in: natural order, out: bit-reversed order. [conv]"""
    a = list(a)
    n = len(a)
    t = n
    m = 1
    while m < n:
        t //= 2
        for i in range(m):
            j1 = 2 * i * t
            S = psirev[m + i]
            for j in range(j1, j1 + t):
                u = a[j]
                v = mulmod(a[j + t], S, q)
                a[j] = (u + v) % q
                a[j + t] = (u - v) % q
    return a


def GS_INTT(a, psiinvrev, q, ninv):
    """Gentleman-Sande DIF, in: bit-reversed order, out: natural order. [conv]"""
    a = list(a)
    n = len(a)
    t = 1
    m = n
    while m > 1:
        j1 = 0
        h = m // 2
        for i in range(h):
            S = psiinvrev[h + i]
            for j in range(j1, j1 + t):
                u = a[j]
                v = a[j + t]
                a[j] = (u + v) % q
                a[j + t] = mulmod((u - v) % q, S, q)
            j1 += 2 * t
        t *= 2
        m //= 2
    return [mulmod(x, ninv, q) for x in a]


def radix4_butterfly(x0, x1, x2, x3, W1, W2, W3, j, q):
    """Fused radix-4 BU (config_radix_selector.radix4_butterfly).
    b'=W1 x1, c'=W2 x2, d'=W3 x3 ; j = order-4 rotator."""
    bp = mulmod(W1, x1, q)
    cp = mulmod(W2, x2, q)
    dp = mulmod(W3, x3, q)
    jb = mulmod(j, bp, q)
    jd = mulmod(j, dp, q)
    y0 = (x0 + bp + cp + dp) % q
    y1 = (x0 + jb - cp - jd) % q
    y2 = (x0 - bp + cp - dp) % q
    y3 = (x0 - jb - cp + jd) % q
    return y0, y1, y2, y3


def dft4(in0, in1, in2, in3, j, q):
    """Independent 4-point DFT with primitive 4th root j (j^2=-1)."""
    j2 = mulmod(j, j, q)      # = -1
    j3 = mulmod(j2, j, q)     # = -j
    pw = [[1, 1, 1, 1],
          [1, j, j2, j3],
          [1, j2, 1, j2],
          [1, j3, j2, j]]
    src = [in0, in1, in2, in3]
    out = []
    for k in range(4):
        acc = 0
        for t in range(4):
            acc = (acc + src[t] * pw[k][t]) % q
        out.append(acc)
    return tuple(out)


# ----------------------------------------------------------------------------
# Host DMA boundary (host_phy) — INVENTED 64b/AXI-Stream framing [source-missing]
# ----------------------------------------------------------------------------
def dma_pack(coeffs):
    """4 x 14-bit coeffs (padded to 16) -> 64-bit little-endian beats."""
    beats = []
    for b in range(0, len(coeffs), 4):
        grp = coeffs[b:b + 4]
        word = 0
        for i, c in enumerate(grp):
            word |= (c & 0x3FFF) << (16 * i)
        beats.append(word)
    return beats


def dma_unpack(beats, count):
    out = []
    for w in beats:
        for i in range(4):
            out.append((w >> (16 * i)) & 0x3FFF)
    return out[:count]


# ----------------------------------------------------------------------------
# Schedule/FSM controller — reconstructed walk [source-missing]
# ----------------------------------------------------------------------------
def fsm_walk(mode_intt: bool, radix: int, n: int):
    states = ["IDLE", "LOAD"]
    stages = (n.bit_length() - 1) if radix == 2 else ((n.bit_length() - 1) // 2)
    for s in range(stages):
        states += ["RUN_STAGE", "NEXT_STAGE"]
    if mode_intt:
        states.append("SCALE")     # multiply by Ninv
    states += ["DRAIN", "DONE", "IDLE"]
    return states


# ============================================================================
# Independent golden (Tier 2) — schoolbook negacyclic mult mod (x^N+1)
# ============================================================================
def negacyclic_mul_golden(a, b, q):
    n = len(a)
    res = [0] * n
    for i in range(n):
        for jj in range(n):
            v = (a[i] * b[jj]) % q
            k = i + jj
            if k < n:
                res[k] = (res[k] + v) % q
            else:
                res[k - n] = (res[k - n] - v) % q
    return res


# ============================================================================
# TIER 1 — size-independent proofs at full bit-width
# ============================================================================
def tier1():
    # --- Barrett correct over ALL x in [0,(q-1)^2], proven symbolically ---
    x = BitVec("x", 64)
    q, mu, k = Q, BARRETT_MU, BARRETT_K
    t = LShR(x * mu, k)
    r0 = x - t * q
    r1 = If(UGE(r0, q), r0 - q, r0)
    r2 = If(UGE(r1, q), r1 - q, r1)
    s = Solver()
    s.add(ULE(x, (Q - 1) * (Q - 1)))
    s.add(Not(And(r2 == URem(x, q), ULT(r2, q))))
    if s.check() == sat:
        m = s.model()
        fail(f"Barrett wrong at x={m[x]} (q={Q}, mu={mu}, k={k})")

    # also confirm <=2 corrections suffice (residual bound) symbolically
    s2 = Solver()
    s2.add(ULE(x, (Q - 1) * (Q - 1)))
    s2.add(UGE(r1, q))             # would a 3rd subtraction ever be needed?
    s2.add(UGE(r1 - q, q))
    if s2.check() == sat:
        fail("Barrett needs >2 conditional subtractions for some x in [0,(q-1)^2]")

    # --- root / scalar constraints (negacyclic) ---
    if pow(PSI, 1024, Q) != Q - 1:
        fail("psi^1024 != -1 mod q (not a negacyclic 2048-th root)")
    if pow(PSI, 2048, Q) != 1:
        fail("psi^2048 != 1 mod q")
    if (PSI * PSI) % Q != OMEGA:
        fail("omega != psi^2 mod q")
    if (N_PROD * NINV_PROD) % Q != 1:
        fail("Ninv is not 1024^-1 mod q")
    if pow(GEN_G, 6, Q) != PSI:
        fail("psi != g^6 mod q (g=11)")
    j = pow(PSI, 512, Q)
    if (j * j) % Q != Q - 1:
        fail("j=psi^512 is not an order-4 root (j^2 != -1)")
    if BARRETT_MU != (1 << BARRETT_K) // Q:
        fail("barrett_mu != floor(2^28/q)")

    # --- conflict-free bank(): butterfly pair (i, i+2^s), bit_s(i)=0, distinct ---
    bits = 16
    iv = BitVec("i", bits)
    def bank_z3(z):
        return (z & 7) ^ (LShR(z, 3) & 7) ^ (LShR(z, 6) & 7) ^ (LShR(z, 9) & 1)
    for sft in range(10):                       # strides 2^0..2^9
        stride = 1 << sft
        sc = Solver()
        sc.add(ULT(iv, N_PROD))
        sc.add(ULT(iv + stride, N_PROD))
        sc.add((LShR(iv, sft) & 1) == 0)        # butterfly low operand: bit s = 0
        sc.add(bank_z3(iv) == bank_z3(iv + stride))
        if sc.check() == sat:
            mm = sc.model()
            fail(f"bank conflict at stride 2^{sft}, i={mm[iv]}")

    # --- (bank,offset) bijection on [0,N) — finite, full index space ---
    seen = set()
    for i in range(N_PROD):
        b, o = bank_fn(i), offset_fn(i)
        if not (0 <= b < BANKS):
            fail(f"bank({i})={b} out of range")
        if not (0 <= o < N_PROD // BANKS):
            fail(f"offset({i})={o} out of range")
        key = (b, o)
        if key in seen:
            fail(f"(bank,offset) not a bijection: collision at i={i} -> {key}")
        seen.add(key)
    if len(seen) != N_PROD:
        fail("(bank,offset) does not cover [0,N)")

    # --- radix-2 butterfly mod-q op (full domain sample) ---
    rng = random.Random(1)
    for _ in range(20000):
        u = rng.randrange(Q); v = rng.randrange(Q); w = rng.randrange(Q)
        vt = mulmod(w, v)
        y0 = (u + vt) % Q; y1 = (u - vt) % Q
        if y0 != (u + w * v) % Q or y1 != (u - w * v) % Q:
            fail("radix-2 butterfly op mismatch")
        if not (0 <= y0 < Q and 0 <= y1 < Q):
            fail("radix-2 butterfly output out of [0,q)")

    # --- radix-4 fusion identity: BU == DFT4 of (x0, W1 x1, W2 x2, W3 x3) ---
    j = pow(PSI, 512, Q)
    for _ in range(20000):
        x0, x1, x2, x3 = (rng.randrange(Q) for _ in range(4))
        W1, W2, W3 = (rng.randrange(Q) for _ in range(3))
        got = radix4_butterfly(x0, x1, x2, x3, W1, W2, W3, j, Q)
        exp = dft4(x0, mulmod(W1, x1), mulmod(W2, x2), mulmod(W3, x3), j, Q)
        if got != exp:
            fail(f"radix-4 fusion != DFT4: {got} vs {exp}")
        if any(not (0 <= y < Q) for y in got):
            fail("radix-4 output out of [0,q)")

    # --- radix-4 op-count reductions (metadata claim) ---
    if R4_MULT_OPT >= R4_MULT_NAIVE or R4_ADDSUB_OPT >= R4_ADDSUB_NAIVE:
        fail("radix-4 optimized counts not below naive")
    if round(100 * (1 - R4_MULT_OPT / R4_MULT_NAIVE)) != 33:
        fail("radix-4 mult reduction != 33%")
    if round(100 * (1 - R4_ADDSUB_OPT / R4_ADDSUB_NAIVE)) != 20:
        fail("radix-4 add/sub reduction != 20%")

    # --- host DMA pack/unpack round-trip (invented boundary) ---
    coeffs = [rng.randrange(1 << COEFF_BITS) for _ in range(N_PROD)]
    if dma_unpack(dma_pack(coeffs), len(coeffs)) != coeffs:
        fail("DMA 64b beat pack/unpack not lossless")

    # --- FSM walk well-formed (terminates at IDLE through DONE) ---
    for mode in (False, True):
        for radix in (2, 4):
            walk = fsm_walk(mode, radix, N_PROD)
            if walk[0] != "IDLE" or walk[-1] != "IDLE" or "DONE" not in walk:
                fail("FSM walk malformed")
            if mode and "SCALE" not in walk:
                fail("INTT FSM walk missing N^-1 SCALE state")
            if (not mode) and "SCALE" in walk:
                fail("NTT FSM walk should skip SCALE")


# ============================================================================
# TIER 2 — whole-system equivalence at N=16 (same prime q), vs schoolbook golden
# ============================================================================
def tier2():
    n = E2E_N
    bits = n.bit_length() - 1
    # negacyclic 2n-th root for n=16: psi_n = psi^(2048/(2n)) = psi^(2048/32)=psi^64
    psi_n = pow(PSI, 2048 // (2 * n), Q)
    if pow(psi_n, n, Q) != Q - 1:
        fail(f"psi_{n}^{n} != -1 (bad reduced negacyclic root)")
    if pow(psi_n, 2 * n, Q) != 1:
        fail(f"psi_{n}^{2*n} != 1")
    psi_inv_n = pow(psi_n, Q - 2, Q)
    ninv_n = pow(n, Q - 2, Q)
    fwd = psi_rev_table(psi_n, n, Q)
    inv = psi_rev_table(psi_inv_n, n, Q)

    rng = random.Random(7)

    # round trip: GS_INTT(CT_NTT(x)) == x
    for _ in range(200):
        xv = [rng.randrange(Q) for _ in range(n)]
        rt = GS_INTT(CT_NTT(xv, fwd, Q), inv, Q, ninv_n)
        if rt != xv:
            fail(f"round trip INTT(NTT(x)) != x: {xv} -> {rt}")

    # full datapath negacyclic mult == independent schoolbook golden
    for _ in range(200):
        a = [rng.randrange(Q) for _ in range(n)]
        b = [rng.randrange(Q) for _ in range(n)]
        A = CT_NTT(a, fwd, Q)
        B = CT_NTT(b, fwd, Q)
        C = [mulmod(A[i], B[i]) for i in range(n)]           # pointwise in bit-rev domain
        got = GS_INTT(C, inv, Q, ninv_n)
        exp = negacyclic_mul_golden(a, b, Q)
        if got != exp:
            fail(f"NTT datapath mult != schoolbook: a={a} b={b}\n got={got}\n exp={exp}")

    # edge cases
    for vec in ([0] * n, [1] + [0] * (n - 1), [Q - 1] * n):
        if GS_INTT(CT_NTT(vec, fwd, Q), inv, Q, ninv_n) != [v % Q for v in vec]:
            fail(f"edge-case round trip failed for {vec}")


# ============================================================================
# Optional deep check — full-size O(N^2) golden, gated OFF by default
# ============================================================================
def deep():
    n = N_PROD
    psi_inv = pow(PSI, Q - 2, Q)
    fwd = psi_rev_table(PSI, n, Q)
    inv = psi_rev_table(psi_inv, n, Q)
    rng = random.Random(99)
    a = [rng.randrange(Q) for _ in range(n)]
    b = [rng.randrange(Q) for _ in range(n)]
    A = CT_NTT(a, fwd, Q); B = CT_NTT(b, fwd, Q)
    C = [mulmod(A[i], B[i]) for i in range(n)]
    got = GS_INTT(C, inv, Q, NINV_PROD)
    exp = negacyclic_mul_golden(a, b, Q)        # O(N^2) at N=1024
    if got != exp:
        fail("DEEP: N=1024 datapath mult != schoolbook golden")


def main():
    tier1()
    tier2()
    if os.environ.get("DEEP_VERIFY") == "1":
        deep()
    print("VERIFIED")


if __name__ == "__main__":
    main()
