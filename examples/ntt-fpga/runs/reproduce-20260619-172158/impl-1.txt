#!/usr/bin/env python3
"""
CFNTT Radix-2/4 NTT Multiplication Accelerator (FPGA) -- reverse implementation
from the scene spec ALONE, with a self-checking golden/SMT verification harness.

Authoritative (from spec): N=1024, q=12289, coeff=14b, product=28b,
psi=1945 (order 2048, negacyclic), omega=psi^2=10302, Ninv=1024^-1=12277,
g=11, j=psi^512, Barrett mu=21843 k=28, radix-4 fused butterfly equations,
pipeline 9 cyc = mult3+addsub1+Barrett4+reg1, FSM, op counts.

Reproduction stand-ins (spec-flagged source-missing): bank XOR-fold map,
bit-reversed twiddle ROM, CT/GS negacyclic butterfly convention, host framing.

Verification strategy: decompose correctness into small finite obligations,
discharge each with the cheapest sound check (z3 over bounded BVs / exhaustive
over a tiny finite space / O(N log N) round-trip).  Prints VERIFIED or FAIL.
"""

import os
import sys
import random

try:
    from z3 import BitVec, BitVecVal, LShR, UGE, ULT, URem, Solver, Or, And, Not, sat
except Exception as e:  # pragma: no cover
    print("FAIL: z3 not available: %r" % (e,))
    sys.exit(1)

# ----------------------------------------------------------------------------
# Authoritative parameters (spec-pinned)
# ----------------------------------------------------------------------------
N        = 1024
LOGN     = 10
Q        = 12289
PSI      = 1945
OMEGA    = 10302
NINV     = 12277
GEN_G    = 11
J_ROT    = pow(PSI, 512, Q)
MU       = 21843
K_SHIFT  = 28
COEFF_W  = 14
PROD_W   = 28
BANKS    = 8
PAR_BU   = 8
PIPE_LAT = 9

def fail(msg):
    print("FAIL: " + msg)
    sys.exit(1)

# ----------------------------------------------------------------------------
# 1. Constant identities (direct arithmetic, cheapest sound check)
# ----------------------------------------------------------------------------
def check_constants():
    if pow(PSI, 2048, Q) != 1:
        fail("psi^2048 != 1 mod q")
    if pow(PSI, 1024, Q) != Q - 1:
        fail("psi^1024 != -1 mod q (not negacyclic): got %d" % pow(PSI, 1024, Q))
    if pow(PSI, 2, Q) != OMEGA:
        fail("omega != psi^2: got %d" % pow(PSI, 2, Q))
    if pow(OMEGA, 1024, Q) != 1 or pow(OMEGA, 512, Q) == 1:
        fail("omega is not a primitive 1024th root")
    if (NINV * N) % Q != 1:
        fail("Ninv*N != 1 mod q")
    if (J_ROT * J_ROT) % Q != Q - 1:
        fail("j-rotator^2 != -1 mod q: j=%d" % J_ROT)
    if pow(GEN_G, 6, Q) != PSI:
        fail("g^6 != psi (generator inconsistent): got %d" % pow(GEN_G, 6, Q))
    if MU != (1 << K_SHIFT) // Q:
        fail("mu != floor(2^k/q)")
    if not (Q * Q < (1 << K_SHIFT)):
        fail("q^2 >= 2^k: Barrett input bound assumption broken")
    if PIPE_LAT != 3 + 1 + 4 + 1:
        fail("pipeline latency decomposition != 9")

# ----------------------------------------------------------------------------
# 2. Barrett modular reduction -- implementation + z3 soundness proof
#    Obligation: for all x in [0, q^2), reduce(x) == x mod q AND in [0,q).
# ----------------------------------------------------------------------------
def barrett_reduce(x):
    # x is a 28-bit product of two reduced operands (x < q^2)
    t = (x * MU) >> K_SHIFT
    r = x - t * Q
    while r >= Q:        # spec: at most 2 conditional subtractions
        r -= Q
    return r

def prove_barrett():
    x = BitVec("x", 64)
    qv  = BitVecVal(Q, 64)
    muv = BitVecVal(MU, 64)
    t  = LShR(x * muv, K_SHIFT)
    r0 = x - t * qv
    r1 = z3_if(UGE(r0, qv), r0 - qv, r0)
    r2 = z3_if(UGE(r1, qv), r1 - qv, r1)
    s = Solver()
    s.add(ULT(x, BitVecVal(Q * Q, 64)))          # bounded, decidable domain
    s.add(Or(r2 != URem(x, qv), UGE(r2, qv)))    # negate the obligation
    if s.check() == sat:
        m = s.model()
        fail("Barrett wrong at x=%s" % m[x])

def z3_if(c, a, b):
    from z3 import If
    return If(c, a, b)

# ----------------------------------------------------------------------------
# 3. Conflict-free memory mapping (committed XOR-fold stand-in)
#    Obligations (exhaustive, finite): (bank,offset) is a bijection on [0,N);
#    every radix-2 butterfly pair and every radix-4 operand quad at every legal
#    stage lands in DISTINCT banks (checked under real integer addressing, NOT
#    the spec's carry-ignoring linear argument).
# ----------------------------------------------------------------------------
def bank(i):
    return (i & 7) ^ ((i >> 3) & 7) ^ ((i >> 6) & 7) ^ ((i >> 9) & 7)

def offset(i):
    return i >> 3

def check_mapping():
    seen = set()
    for i in range(N):
        key = (bank(i), offset(i))
        if key in seen:
            fail("bank/offset not a bijection: collision %r at i=%d" % (key, i))
        seen.add(key)
        if not (0 <= bank(i) < BANKS) or not (0 <= offset(i) < N // BANKS):
            fail("bank/offset out of range at i=%d" % i)
    if len(seen) != N:
        fail("bank/offset does not cover N words")

    # radix-2: pairs (i, i+2^s) with bit s of i == 0
    for s in range(LOGN):
        stride = 1 << s
        for i in range(N):
            if (i >> s) & 1:
                continue
            if bank(i) == bank(i + stride):
                fail("radix-2 bank conflict stride 2^%d at i=%d" % (s, i))

    # radix-4: quad (i, i+2^s, i+2^(s+1), i+3*2^s) with bits s,s+1 of i == 0
    for s in range(LOGN - 1):
        st = 1 << s
        for i in range(N):
            if ((i >> s) & 1) or ((i >> (s + 1)) & 1):
                continue
            quad = (i, i + st, i + 2 * st, i + 3 * st)
            if max(quad) >= N:
                continue
            bset = {bank(j) for j in quad}
            if len(bset) != 4:
                fail("radix-4 bank conflict stride 2^%d at i=%d -> banks %r"
                     % (s, i, sorted(bank(j) for j in quad)))

# ----------------------------------------------------------------------------
# 4. Negacyclic NTT / INTT (merged psi-twist, CT forward / GS inverse).
#    Reference (golden) negacyclic convolution is the INDEPENDENT model.
# ----------------------------------------------------------------------------
def bitrev(x, bits):
    r = 0
    for _ in range(bits):
        r = (r << 1) | (x & 1)
        x >>= 1
    return r

PSI_INV = pow(PSI, Q - 2, Q)
_psi_pows     = [pow(PSI, i, Q) for i in range(N)]
_psi_inv_pows = [pow(PSI_INV, i, Q) for i in range(N)]
PSI_BR     = [_psi_pows[bitrev(i, LOGN)] for i in range(N)]
PSI_INV_BR = [_psi_inv_pows[bitrev(i, LOGN)] for i in range(N)]

def ntt(a):
    a = list(a)
    n = len(a); t = n; m = 1
    while m < n:
        t //= 2
        for i in range(m):
            j1 = 2 * i * t
            S = PSI_BR[m + i]
            for j in range(j1, j1 + t):
                U = a[j]
                V = a[j + t] * S % Q
                a[j] = (U + V) % Q
                a[j + t] = (U - V) % Q
        m *= 2
    return a

def intt(a):
    a = list(a)
    n = len(a); t = 1; m = n
    while m > 1:
        j1 = 0; h = m // 2
        for i in range(h):
            S = PSI_INV_BR[h + i]
            for j in range(j1, j1 + t):
                U = a[j]
                V = a[j + t]
                a[j] = (U + V) % Q
                a[j + t] = (U - V) * S % Q
            j1 += 2 * t
        t *= 2; m //= 2
    return [x * NINV % Q for x in a]

def negacyclic_conv_golden(a, b):
    # c = a*b mod (x^N + 1), mod q  -- independent O(N^2) reference
    n = len(a)
    c = [0] * n
    for i in range(n):
        ai = a[i]
        if ai == 0:
            continue
        for k in range(n):
            p = ai * b[k]
            idx = i + k
            if idx < n:
                c[idx] = (c[idx] + p) % Q
            else:
                c[idx - n] = (c[idx - n] - p) % Q
    return [x % Q for x in c]

def check_ntt():
    rng = random.Random(20220101)
    # round-trip identity: INTT(NTT(x)) == x
    for _ in range(4):
        x = [rng.randrange(Q) for _ in range(N)]
        if intt(ntt(x)) != x:
            fail("INTT(NTT(x)) != x")
    # NTT(0)=0, NTT(constant) sanity
    if ntt([0] * N) != [0] * N:
        fail("NTT(0) != 0")
    # negacyclic convolution via NTT == golden (one full-size pair)
    a = [rng.randrange(Q) for _ in range(N)]
    b = [rng.randrange(Q) for _ in range(N)]
    fa, fb = ntt(a), ntt(b)
    cc = intt([x * y % Q for x, y in zip(fa, fb)])
    if cc != negacyclic_conv_golden(a, b):
        fail("NTT-based negacyclic convolution != golden")

# ----------------------------------------------------------------------------
# 5. Radix-4 fused butterfly: spec equations vs a two-radix-2-stage
#    decomposition.  Proven as a ring identity by EXHAUSTIVE evaluation over a
#    tiny field Z_5 (j=2, 2^2=-1): both sides are multilinear in x0..x3 (deg 1)
#    and degree<=3 in w, and |Z_5|>3, so agreement on all of Z_5^5 forces the
#    polynomial identity over Z, hence over Z_q for any j with j^2=-1.
# ----------------------------------------------------------------------------
def radix4_spec(x0, x1, x2, x3, w, j, m):
    W1 = w % m
    W2 = w * w % m
    W3 = w * w * w % m
    bp = W1 * x1 % m
    cp = W2 * x2 % m
    dp = W3 * x3 % m
    y0 = (x0 + bp + cp + dp) % m
    y1 = (x0 + j * bp - cp - j * dp) % m
    y2 = (x0 - bp + cp - dp) % m
    y3 = (x0 - j * bp - cp + j * dp) % m
    return y0, y1, y2, y3

def radix4_two_stage(x0, x1, x2, x3, w, j, m):
    W1 = w % m
    W2 = w * w % m
    # stage 1: bf(x0,x2;W2), bf(x1,x3;W2)
    e0 = (x0 + W2 * x2) % m
    e1 = (x0 - W2 * x2) % m
    f0 = (x1 + W2 * x3) % m
    f1 = (x1 - W2 * x3) % m
    # stage 2: bf(e0,f0;W1) -> y0,y2 ; bf(e1,f1;j*W1) -> y1,y3
    y0 = (e0 + W1 * f0) % m
    y2 = (e0 - W1 * f0) % m
    y1 = (e1 + (j * W1) * f1) % m
    y3 = (e1 - (j * W1) * f1) % m
    return y0, y1, y2, y3

def check_radix4():
    m = 5; j = 2
    if (j * j) % m != m - 1:
        fail("test-field j^2 != -1")
    for x0 in range(m):
        for x1 in range(m):
            for x2 in range(m):
                for x3 in range(m):
                    for w in range(m):
                        if radix4_spec(x0, x1, x2, x3, w, j, m) != \
                           radix4_two_stage(x0, x1, x2, x3, w, j, m):
                            fail("radix-4 != two radix-2 stages at "
                                 "(%d,%d,%d,%d,w=%d)" % (x0, x1, x2, x3, w))
    # confirm the real rotator satisfies the relation the identity assumes
    if (J_ROT * J_ROT) % Q != Q - 1:
        fail("real j-rotator not order-4")

# ----------------------------------------------------------------------------
# 6. Op-count claims (within-radix-4 baseline, per spec reconciliation)
# ----------------------------------------------------------------------------
def check_opcounts():
    r4_mult_naive, r4_mult_opt = 3, 2
    r4_as_naive,  r4_as_opt    = 10, 8
    if round(100 * (1 - r4_mult_opt / r4_mult_naive)) != 33:
        fail("mult reduction != 33%")
    if round(100 * (1 - r4_as_opt / r4_as_naive)) != 20:
        fail("add/sub reduction != 20%")
    # cycle accounting
    if 5 != (N.bit_length() - 1) // 2:  # log4(1024)=5
        fail("radix-4 stage count != 5")
    if 256 != N // 4:
        fail("radix-4 butterflies/stage != 256")
    if 32 != (N // 4) // PAR_BU:
        fail("radix-4 cycles/stage != 32 (divide by 8 BUs, not 4 lanes)")
    if 160 != 5 * ((N // 4) // PAR_BU):
        fail("radix-4 total butterfly cycles != 160")

# ----------------------------------------------------------------------------
# 7. FSM well-formedness (reconstructed controller)
# ----------------------------------------------------------------------------
FSM = {
    "IDLE":        ["LOAD"],
    "LOAD":        ["RUN_STAGES"],
    "RUN_STAGES":  ["RUN_STAGES", "SCALE_INTT", "DRAIN"],
    "SCALE_INTT":  ["DRAIN"],
    "DRAIN":       ["DONE"],
    "DONE":        ["IDLE"],
}

def check_fsm():
    states = set(FSM)
    for s, outs in FSM.items():
        for d in outs:
            if d not in states:
                fail("FSM transition to unknown state %s" % d)
    # reachability from IDLE
    seen = set(); stack = ["IDLE"]
    while stack:
        s = stack.pop()
        if s in seen:
            continue
        seen.add(s)
        stack.extend(FSM[s])
    if seen != states:
        fail("FSM has unreachable states: %r" % (states - seen))
    # every state can reach DONE (no dead ends)
    def reaches(target):
        ok = set()
        changed = True
        while changed:
            changed = False
            for s in states:
                if s in ok:
                    continue
                if target in FSM[s] or any(d in ok for d in FSM[s]):
                    ok.add(s); changed = True
        return ok
    if reaches("DONE") != states:
        fail("some FSM states cannot reach DONE")
    if "DONE" not in FSM and True:
        pass
    if "IDLE" not in FSM["DONE"]:
        fail("DONE does not return to IDLE (dangling)")

# ----------------------------------------------------------------------------
# 8. Host AXI-Stream beat packing (reproduction-invented boundary)
#    4 coeffs (14b padded to 16b) per 64-bit beat; pack/unpack round-trip.
# ----------------------------------------------------------------------------
def pack_beats(coeffs):
    beats = []
    for i in range(0, len(coeffs), 4):
        grp = coeffs[i:i + 4]
        beat = 0
        for k, c in enumerate(grp):
            if not (0 <= c < (1 << COEFF_W)):
                fail("coefficient out of 14-bit range: %d" % c)
            beat |= (c & 0xFFFF) << (16 * k)
        beats.append(beat)
    return beats

def unpack_beats(beats, count):
    out = []
    for beat in beats:
        for k in range(4):
            out.append((beat >> (16 * k)) & ((1 << COEFF_W) - 1))
    return out[:count]

def check_host():
    rng = random.Random(7)
    coeffs = [rng.randrange(Q) for _ in range(N)]
    if unpack_beats(pack_beats(coeffs), N) != coeffs:
        fail("host beat pack/unpack round-trip mismatch")
    if N % 4 != 0:
        fail("N not a multiple of coeffs/beat")

# ----------------------------------------------------------------------------
# 9. INTT N^-1 scaling unit + modular multiply via Barrett (spot identity)
# ----------------------------------------------------------------------------
def modmul_barrett(a, b):
    return barrett_reduce((a % Q) * (b % Q))

def check_modmul_and_scale():
    rng = random.Random(99)
    for _ in range(2000):
        a = rng.randrange(Q); b = rng.randrange(Q)
        if modmul_barrett(a, b) != (a * b) % Q:
            fail("Barrett modmul wrong at a=%d b=%d" % (a, b))
    # INTT scale step y = x*Ninv mod q reverses an N-multiply
    for _ in range(1000):
        x = rng.randrange(Q)
        if (modmul_barrett(x * N % Q, NINV)) != x:
            fail("Ninv scaling does not invert N at x=%d" % x)

# ----------------------------------------------------------------------------
# main
# ----------------------------------------------------------------------------
def main():
    check_constants()
    prove_barrett()
    check_mapping()
    check_modmul_and_scale()
    check_radix4()
    check_opcounts()
    check_fsm()
    check_host()
    check_ntt()  # heaviest (one full-size negacyclic conv); last

    if os.environ.get("DEEP_VERIFY") == "1":
        rng = random.Random(424242)
        for _ in range(20):
            a = [rng.randrange(Q) for _ in range(N)]
            b = [rng.randrange(Q) for _ in range(N)]
            cc = intt([x * y % Q for x, y in zip(ntt(a), ntt(b))])
            if cc != negacyclic_conv_golden(a, b):
                fail("DEEP: negacyclic conv mismatch")

    print("VERIFIED")
    sys.exit(0)

if __name__ == "__main__":
    main()
