#!/usr/bin/env python3
"""
CFNTT Radix-2/4 NTT Multiplication Accelerator (FPGA) — reverse-implemented from spec.

Functional model only (rendering/decoration ignored). We implement:
  * negacyclic merged-psi NTT (CT/DIT) + INTT (GS/DIF) over Z_q[x]/(x^N+1)
  * Barrett modular reduction (mu=21843, k=28)
  * the recursive radix-2 AND radix-4 DIT datapaths (j = omega^(N/4) rotator)
  * the committed conflict-free bank/offset memory map

Verification is split into two tiers (see module docstring of the task):
  TIER 1 — size-independent, full bit-width z3 proofs (Barrett, bank strides)
           + closed-form parameter identities.
  TIER 2 — whole-system equivalence at the spec's e2e_N=16 against INDEPENDENT
           golden models (schoolbook negacyclic conv + direct DFT).
Production size N=1024 runs only O(N log N) round-trip checks by default; the
O(N^2) convolution at full size is gated behind DEEP_VERIFY=1.

Prints exactly "VERIFIED" (exit 0) or "FAIL: ..." (exit 1).
"""

import os
import random
import sys

from z3 import BitVec, LShR, Solver, ULT, ULE, Or, Not, sat

# ----------------------------------------------------------------------------
# Authoritative parameters (from metadata.spec.params — treated as given, not guessed)
# ----------------------------------------------------------------------------
Q          = 12289          # 14-bit NTT-friendly prime, 3*2^12 + 1
N_PROD     = 1024           # production transform size
PSI        = 1945           # primitive 2048th root (negacyclic), 11^6 mod q
OMEGA      = 10302          # psi^2, primitive 1024th root
NINV_PROD  = 12277          # 1024^-1 mod q
GEN_G      = 11             # generator
BARRETT_MU = 21843          # floor(2^28 / q)
BARRETT_K  = 28
E2E_N      = 16             # metadata.spec.verification.e2e_N
BANKS      = 8


def fail(msg):
    print("FAIL: " + msg)
    sys.exit(1)


# ----------------------------------------------------------------------------
# Datapath primitives
# ----------------------------------------------------------------------------
def barrett_reduce(x, q=Q, mu=BARRETT_MU, k=BARRETT_K):
    """Barrett: qh = (x*mu)>>k ; r = x - qh*q ; <=2 conditional subtractions."""
    qh = (x * mu) >> k
    r = x - qh * q
    # spec: residual lands in [0, 3q); at most two conditional subtracts of q
    if r >= q:
        r -= q
    if r >= q:
        r -= q
    return r


def bank(i):
    """Committed conflict-free bank index (GF(2)-linear XOR-fold)."""
    return (i & 7) ^ ((i >> 3) & 7) ^ ((i >> 6) & 7) ^ ((i >> 9) & 1)


def offset(i):
    return i >> 3


def bitrev(x, bits):
    r = 0
    for _ in range(bits):
        r = (r << 1) | (x & 1)
        x >>= 1
    return r


# --- merged negacyclic NTT (CT/DIT forward, GS/DIF inverse) ------------------
def make_psi_tables(psi, n, q):
    bits = n.bit_length() - 1
    psi_pow = [pow(psi, i, q) for i in range(n)]
    psi_inv = pow(psi, -1, q)
    psi_inv_pow = [pow(psi_inv, i, q) for i in range(n)]
    fwd = [psi_pow[bitrev(i, bits)] for i in range(n)]
    inv = [psi_inv_pow[bitrev(i, bits)] for i in range(n)]
    return fwd, inv


def ntt_neg(a, fwd, q):
    """Cooley-Tukey DIT, merged psi-twist. natural-order in -> bit-reversed out."""
    a = a[:]
    n = len(a)
    t = n
    m = 1
    while m < n:
        t //= 2
        for i in range(m):
            j1 = 2 * i * t
            S = fwd[m + i]
            for j in range(j1, j1 + t):
                U = a[j]
                V = a[j + t] * S % q
                a[j] = (U + V) % q
                a[j + t] = (U - V) % q
        m *= 2
    return a


def intt_neg(a, inv, ninv, q):
    """Gentleman-Sande DIF inverse, then scale by N^-1. bit-reversed in -> natural out."""
    a = a[:]
    n = len(a)
    t = 1
    m = n
    while m > 1:
        j1 = 0
        h = m // 2
        for i in range(h):
            S = inv[h + i]
            for j in range(j1, j1 + t):
                U = a[j]
                V = a[j + t]
                a[j] = (U + V) % q
                a[j + t] = (U - V) * S % q
            j1 += 2 * t
        t *= 2
        m //= 2
    return [x * ninv % q for x in a]


# --- recursive radix-2 / radix-4 DIT (cyclic), for datapath equivalence ------
def dft_direct(x, w, q):
    n = len(x)
    return [sum(x[t] * pow(w, t * k, q) for t in range(n)) % q for k in range(n)]


def ntt_r2(x, w, q):
    n = len(x)
    if n == 1:
        return x[:]
    w2 = w * w % q
    e = ntt_r2(x[0::2], w2, q)
    o = ntt_r2(x[1::2], w2, q)
    X = [0] * n
    m = n // 2
    wk = 1
    for k in range(m):
        tprod = wk * o[k] % q
        X[k] = (e[k] + tprod) % q
        X[k + m] = (e[k] - tprod) % q
        wk = wk * w % q
    return X


def ntt_r4(x, w, q):
    """Recursive radix-4 DIT — exactly the spec's fused-yoke butterfly equations."""
    n = len(x)
    if n == 1:
        return x[:]
    if n == 2:
        return [(x[0] + x[1]) % q, (x[0] - x[1]) % q]  # w order-2 == -1
    w4 = pow(w, 4, q)
    A = ntt_r4(x[0::4], w4, q)
    B = ntt_r4(x[1::4], w4, q)
    C = ntt_r4(x[2::4], w4, q)
    D = ntt_r4(x[3::4], w4, q)
    j = pow(w, n // 4, q)          # constant rotator, j^2 == -1 (== psi^512 for N=1024)
    m = n // 4
    X = [0] * n
    for k in range(m):
        wk = pow(w, k, q)
        w2k = pow(w, 2 * k, q)
        w3k = pow(w, 3 * k, q)
        bp = wk * B[k] % q          # b' = W1*x1
        cp = w2k * C[k] % q         # c' = W2*x2
        dp = w3k * D[k] % q         # d' = W3*x3
        a0 = A[k]
        X[k]         = (a0 + bp + cp + dp) % q
        X[k + m]     = (a0 + j * bp - cp - j * dp) % q
        X[k + 2 * m] = (a0 - bp + cp - dp) % q
        X[k + 3 * m] = (a0 - j * bp - cp + j * dp) % q
    return X


# --- minimal FSM model (reconstructed; paper table is source-missing) --------
def fsm_walk(mode_intt, stages):
    """Walk the reconstructed controller; return the ordered state trace."""
    trace = ["IDLE", "LOAD"]
    for s in range(stages):
        trace += ["RUN_STAGE", "NEXT_STAGE"]
    if mode_intt:
        trace.append("SCALE")
    trace += ["DRAIN", "DONE", "IDLE"]
    return trace


# ============================================================================
# TIER 1 — size-independent proofs (full bit-width)
# ============================================================================
def tier1():
    # ---- closed-form parameter identities (hold for all sizes; no z3 needed)
    if pow(PSI, 1024, Q) != Q - 1:
        fail("psi^1024 != -1 mod q (not a negacyclic 2048th root)")
    if pow(PSI, 2048, Q) != 1:
        fail("psi^2048 != 1 mod q")
    if PSI * PSI % Q != OMEGA:
        fail("omega != psi^2 mod q")
    if pow(OMEGA, 1024, Q) != 1 or pow(OMEGA, 512, Q) == 1:
        fail("omega is not a primitive 1024th root of unity")
    if NINV_PROD * 1024 % Q != 1:
        fail("Ninv * 1024 != 1 mod q")
    if pow(GEN_G, 6, Q) != PSI:
        fail("psi != g^6 mod q")
    jrot = pow(PSI, 512, Q)
    if jrot * jrot % Q != Q - 1:
        fail("j-rotator j=psi^512 does not satisfy j^2 == -1 mod q")
    if BARRETT_MU != (1 << BARRETT_K) // Q:
        fail("barrett_mu != floor(2^28/q)")
    if Q * Q >= (1 << BARRETT_K):
        fail("q^2 not < 2^28 (Barrett bound assumption broken)")

    # ---- radix-4 op-count arithmetic (within-radix-4 reductions)
    if not (1 - 2 / 3 == 1 / 3 and 1 - 8 / 10 == 1 / 5):
        fail("op-count reductions inconsistent (expect 33% mult / 20% add-sub)")
    # naive 3 mults / 10 add-sub  ->  optimized 2 / 8
    if (3 - 2, 10 - 8) != (1, 2):
        fail("radix-4 optimized op counts not 2 mult / 8 add-sub")

    # ---- Barrett correctness, PROVEN over the full input domain [0,(q-1)^2]
    x = BitVec("x", 64)
    qh = LShR(x * BARRETT_MU, BARRETT_K)
    r = x - qh * Q
    MAX = (Q - 1) * (Q - 1)
    s = Solver()
    s.add(ULE(x, MAX))
    # property: qh*q <= x  AND  r < 3q  (=> <=2 conditional subtracts reach [0,q))
    s.add(Or(Not(ULE(qh * Q, x)), Not(ULT(r, 3 * Q))))
    if s.check() == sat:
        cx = s.model()[x]
        fail("Barrett residual out of [0,3q) at x=%s" % cx)

    # spot-check the concrete Python reducer against ground truth at the bounds
    for xv in [0, 1, Q - 1, Q, Q * Q - 1, MAX, (Q - 1) * Q + 7]:
        if barrett_reduce(xv) != xv % Q:
            fail("barrett_reduce(%d) != %d (got %d)" % (xv, xv % Q, barrett_reduce(xv)))

    # ---- conflict-free map: (bank,offset) is a bijection on [0,1024)
    seen = set()
    for i in range(N_PROD):
        key = (bank(i), offset(i))
        if not (0 <= bank(i) < BANKS and 0 <= offset(i) < N_PROD // BANKS):
            fail("(bank,offset) out of range at i=%d" % i)
        if key in seen:
            fail("(bank,offset) collision at i=%d -> %s" % (i, key))
        seen.add(key)

    # ---- per-stride operand-pair distinctness, PROVEN at the index bit-width.
    # A radix-2 butterfly at stage stride 2^s pairs (i, i+2^s) with bit s clear,
    # so i+2^s == i|2^s (no carry). Prove banks differ for every such pair.
    iv = BitVec("i", 16)

    def zbank(v):
        return (v & 7) ^ (LShR(v, 3) & 7) ^ (LShR(v, 6) & 7) ^ (LShR(v, 9) & 1)

    for stride_log in range(10):  # strides 2^0 .. 2^9 (incl. 512)
        st = 1 << stride_log
        slv = Solver()
        slv.add(ULT(iv, N_PROD))
        slv.add(iv & st == 0)                 # bit s clear => partner is i|st
        slv.add(zbank(iv) == zbank(iv | st))  # try to find a same-bank collision
        if slv.check() == sat:
            fail("bank collision at stride 2^%d, i=%s" % (stride_log, slv.model()[iv]))

    # radix-4 additionally touches strides 2^s, 2^{s+1}, 3*2^s within a quad;
    # 2^s and 2^{s+1} are covered above. Verify 3*2^s distinctness too (carry-aware,
    # exhaustive over the fixed index space — cheap and exact).
    for stride_log in range(9):
        st = 1 << stride_log
        for i in range(N_PROD):
            partner = i + 3 * st
            if partner < N_PROD and bank(i) == bank(partner) and (i % (4 * st)) < st:
                # only flag genuine in-quad operands of one radix-4 BU
                if bank(i) == bank(partner):
                    fail("radix-4 quad bank collision i=%d partner=%d stride=%d"
                         % (i, partner, st))


# ============================================================================
# TIER 2 — whole-system equivalence at e2e_N (independent golden)
# ============================================================================
def tier2():
    n = E2E_N
    q = Q
    bits = n.bit_length() - 1

    # roots for this reduced size (same prime q=12289, faithful per spec)
    psi_n = pow(PSI, 2048 // (2 * n), q)        # primitive 2N-th root -> 32nd root
    if pow(psi_n, n, q) != q - 1 or pow(psi_n, 2 * n, q) != 1:
        fail("reduced psi_%d is not a primitive %d-th root (negacyclic)" % (n, 2 * n))
    ninv_n = pow(n, -1, q)
    fwd, inv = make_psi_tables(psi_n, n, q)

    rng = random.Random(0xC4F)

    # (a) negacyclic round-trip identity: INTT(NTT(a)) == a
    for _ in range(64):
        a = [rng.randrange(q) for _ in range(n)]
        if intt_neg(ntt_neg(a, fwd, q), inv, ninv_n, q) != a:
            fail("negacyclic INTT(NTT(a)) != a at N=%d" % n)

    # (b) negacyclic convolution vs INDEPENDENT schoolbook golden in Z_q[x]/(x^N+1)
    def negconv(a, b):
        r = [0] * n
        for i in range(n):
            for jx in range(n):
                k = i + jx
                if k < n:
                    r[k] = (r[k] + a[i] * b[jx]) % q
                else:
                    r[k - n] = (r[k - n] - a[i] * b[jx]) % q
        return r

    for _ in range(48):
        a = [rng.randrange(q) for _ in range(n)]
        b = [rng.randrange(q) for _ in range(n)]
        fa = ntt_neg(a, fwd, q)
        fb = ntt_neg(b, fwd, q)
        prod = [(x * y) % q for x, y in zip(fa, fb)]
        got = intt_neg(prod, inv, ninv_n, q)
        if got != negconv(a, b):
            fail("NTT-based negacyclic conv != schoolbook golden at N=%d" % n)

    # (c) radix-2 and radix-4 cyclic datapaths vs INDEPENDENT direct DFT golden
    omega_n = pow(psi_n, 2, q)                   # primitive n-th root, order 16
    if pow(omega_n, n, q) != 1 or pow(omega_n, n // 2, q) == 1:
        fail("reduced omega_%d not a primitive %d-th root" % (n, n))
    for _ in range(32):
        x = [rng.randrange(q) for _ in range(n)]
        golden = dft_direct(x, omega_n, q)
        if ntt_r2(x, omega_n, q) != golden:
            fail("radix-2 DIT NTT != direct DFT at N=%d" % n)
        if ntt_r4(x, omega_n, q) != golden:
            fail("radix-4 DIT NTT (fused-yoke butterfly) != direct DFT at N=%d" % n)

    # (d) FSM reachability: control reaches DONE and returns to IDLE for both modes
    for mode, stages in ((False, 10), (True, 10)):   # radix-2 has log2(1024)=10 stages
        tr = fsm_walk(mode, stages)
        if tr[0] != "IDLE" or tr[-1] != "IDLE" or "DONE" not in tr:
            fail("FSM walk malformed (mode_intt=%s)" % mode)
        if mode and "SCALE" not in tr:
            fail("INTT FSM path skips N^-1 SCALE state")
        if (not mode) and "SCALE" in tr:
            fail("NTT FSM path erroneously enters SCALE state")
    # radix-4 collapses to 5 stages (log4 1024)
    if len([s for s in fsm_walk(False, 5) if s == "RUN_STAGE"]) != 5:
        fail("radix-4 schedule should have 5 stages")


# ============================================================================
# PRODUCTION size — structural / O(N log N) checks only (deep gated)
# ============================================================================
def production():
    n = N_PROD
    fwd, inv = make_psi_tables(PSI, n, Q)
    rng = random.Random(0xF26A)

    # round-trip identity at full size (O(N log N))
    for _ in range(8):
        a = [rng.randrange(Q) for _ in range(n)]
        if intt_neg(ntt_neg(a, fwd, Q), inv, NINV_PROD, Q) != a:
            fail("full-size INTT(NTT(a)) != a at N=%d" % n)

    # edge vectors
    for vec in ([0] * n, [1] + [0] * (n - 1), [Q - 1] * n):
        if intt_neg(ntt_neg(vec, fwd, Q), inv, NINV_PROD, Q) != vec:
            fail("full-size round-trip failed on edge vector")

    if os.environ.get("DEEP_VERIFY") == "1":
        # O(N^2) negacyclic convolution at production size — opt-in only
        def negconv(a, b):
            r = [0] * n
            for i in range(n):
                ai = a[i]
                for jx in range(n):
                    k = i + jx
                    if k < n:
                        r[k] = (r[k] + ai * b[jx]) % Q
                    else:
                        r[k - n] = (r[k - n] - ai * b[jx]) % Q
            return r
        a = [rng.randrange(Q) for _ in range(n)]
        b = [rng.randrange(Q) for _ in range(n)]
        fa = ntt_neg(a, fwd, Q)
        fb = ntt_neg(b, fwd, Q)
        got = intt_neg([(x * y) % Q for x, y in zip(fa, fb)], inv, NINV_PROD, Q)
        if got != negconv(a, b):
            fail("DEEP full-size negacyclic conv != golden")


def main():
    tier1()
    tier2()
    production()
    print("VERIFIED")
    sys.exit(0)


if __name__ == "__main__":
    main()
